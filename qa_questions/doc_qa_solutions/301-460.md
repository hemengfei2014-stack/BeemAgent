# 问题 301-460: 特定文档详情、标题/内容长度、时间范围统计

## 问题描述分类
- 问题 301-310: MCN文档详情
- 问题 311-340: 标题最长的文档
- 问题 321-330: 内容最长的文档
- 问题 331-340: 标题最长的前5个文档
- 问题 341-350: 所有文档的创建时间
- 问题 351-370: 按年份统计文档
- 问题 371-380: 最早创建的文档
- 问题 381-399: 7天/30天内创建的文档

## 代码实现

```python
from utils import doc_loader, CURRENT_USER_ID
from datetime import datetime, timedelta

print("=" * 60)
print("文档系统查询 - 文档详情、长度统计、时间范围")
print("=" * 60)

# ============ MCN文档详情 (问题 301-310) ============
print("\n【问题 301-310】MCN文档详情")

mcn_docs = doc_loader.search_docs_by_keyword("MCN", CURRENT_USER_ID)

print(f"\n标题含'MCN'的文档: {len(mcn_docs)}个")
for doc in mcn_docs:
    print(f"\n文档标题: {doc.title}")
    print(f"文档ID: {doc.doc_id}")
    print(f"类型: {doc_loader.get_doc_type_name(doc.ctype)}")
    print(f"创建者: {doc_loader.get_user_name(doc.owner)}")
    print(f"创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")
    print(f"完整内容:\n{doc.note}")

# ============ 标题/内容长度统计 (问题 311-340) ============
print("\n=== 标题/内容长度统计 ===")

accessible_docs = doc_loader.get_user_docs(CURRENT_USER_ID)

if accessible_docs:
    # 标题长度排序
    by_title_len = sorted(accessible_docs, key=lambda x: len(x.title), reverse=True)

    print("\n标题最长的文档:")
    longest_title = by_title_len[0]
    print(f"  标题: {longest_title.title}")
    print(f"  标题长度: {len(longest_title.title)}字符")
    print(f"  创建者: {doc_loader.get_user_name(longest_title.owner)}")

    print("\n标题最长的前5个文档:")
    for i, doc in enumerate(by_title_len[:5], 1):
        print(f"  {i}. {doc.title} ({len(doc.title)}字符)")

    # 内容长度排序
    by_content_len = sorted(accessible_docs, key=lambda x: len(x.note), reverse=True)

    print("\n内容最长的文档:")
    longest_content = by_content_len[0]
    print(f"  标题: {longest_content.title}")
    print(f"  内容长度: {len(longest_content.note)}字符")

    print("\n内容最长的前5个文档:")
    for i, doc in enumerate(by_content_len[:5], 1):
        print(f"  {i}. {doc.title} ({len(doc.note)}字符)")
else:
    print("没有可访问的文档")

# ============ 所有文档的创建时间 (问题 341-350) ============
print("\n=== 所有文档的创建时间 ===")

if accessible_docs:
    print("\n文档及其创建时间:")
    for doc in accessible_docs:
        time_str = doc_loader.timestamp_to_datetime(doc.create_time)
        print(f"  {doc.title}: {time_str}")

# ============ 按年份统计 (问题 351-370) ============
print("\n=== 按年份统计 ===")

year_counts = {}
for doc in accessible_docs:
    year = datetime.fromtimestamp(doc.create_time).year
    year_counts[year] = year_counts.get(year, 0) + 1

print("\n按年份统计:")
for year in sorted(year_counts.keys()):
    print(f"  {year}年: {year_counts[year]}个")

# 列出每年的文档
for year in sorted(year_counts.keys()):
    print(f"\n{year}年创建的文档:")
    year_docs = [doc for doc in accessible_docs
                 if datetime.fromtimestamp(doc.create_time).year == year]
    for doc in year_docs:
        print(f"  - {doc.title}")

# ============ 最早/最新创建的文档 (问题 371-380) ============
print("\n=== 最早/最新创建的文档 ===")

if accessible_docs:
    sorted_by_time = sorted(accessible_docs, key=lambda x: x.create_time)

    earliest = sorted_by_time[0]
    latest = sorted_by_time[-1]

    print(f"\n最早创建: {earliest.title}")
    print(f"  时间: {doc_loader.timestamp_to_datetime(earliest.create_time)}")

    print(f"\n最新创建: {latest.title}")
    print(f"  时间: {doc_loader.timestamp_to_datetime(latest.create_time)}")

# ============ 7天/30天内创建的文档 (问题 381-399) ============
print("\n=== 7天/30天内创建的文档 ===")

now = datetime.now()
week_ago_ts = int((now - timedelta(days=7)).timestamp())
month_ago_ts = int((now - timedelta(days=30)).timestamp())

recent_week = [doc for doc in accessible_docs if doc.create_time >= week_ago_ts]
recent_month = [doc for doc in accessible_docs if doc.create_time >= month_ago_ts]

print(f"\n最近7天创建: {len(recent_week)}个")
if recent_week:
    for doc in recent_week:
        print(f"  - {doc.title}")

print(f"\n最近30天创建: {len(recent_month)}个")
if recent_month:
    for doc in recent_month:
        print(f"  - {doc.title}")
else:
    print("  （文档创建时间较早，最近30天无新文档）")

# ============ 文档统计汇总 ============
print("\n=== 文档统计汇总 ===")

print(f"\n可访问文档总数: {len(accessible_docs)}")
print(f"文档创建时间范围:")
if accessible_docs:
    times = [doc.create_time for doc in accessible_docs]
    print(f"  最早: {doc_loader.timestamp_to_datetime(min(times))}")
    print(f"  最新: {doc_loader.timestamp_to_datetime(max(times))}")

print("\n" + "=" * 60)
```
