# 问题 1-50: 可访问文档列表、我创建的文档

## 问题描述
- 问题 1-10: 我能访问哪些文档
- 问题 11-30: 我最近创建的文档
- 问题 31-50: 我最新创建的文档

## 代码实现

```python
from utils import doc_loader, CURRENT_USER_ID

print("=" * 60)
print(f"文档系统查询 - 当前用户: {doc_loader.get_user_name(CURRENT_USER_ID)} (ID:{CURRENT_USER_ID})")
print("=" * 60)

# ============ 我能访问的文档 (问题 1-10) ============
print("\n【问题 1-10】我能访问的文档")

accessible_docs = doc_loader.get_user_docs(CURRENT_USER_ID)
print(f"\n我可访问的文档数量: {len(accessible_docs)}")

if accessible_docs:
    print("\n文档列表:")
    for i, doc in enumerate(accessible_docs, 1):
        print(f"  {i}. [{doc_loader.get_doc_type_name(doc.ctype)}] {doc.title}")
        print(f"     ID: {doc.doc_id}")
        print(f"     创建者: {doc_loader.get_user_name(doc.owner)}")
        print(f"     创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")
        print(f"     备注: {doc.note[:50]}..." if len(doc.note) > 50 else f"     备注: {doc.note}")
        print()
else:
    print("没有可访问的文档")

# ============ 我创建的文档 (问题 11-30) ============
print("\n【问题 11-30】我创建的文档")

created_docs = doc_loader.get_user_created_docs(CURRENT_USER_ID)
print(f"\n我创建的文档数量: {len(created_docs)}")

if created_docs:
    # 按创建时间排序（从新到旧）
    created_docs_sorted = sorted(created_docs, key=lambda x: x.create_time, reverse=True)

    print("\n文档列表（按创建时间从新到旧）:")
    for i, doc in enumerate(created_docs_sorted, 1):
        print(f"  {i}. [{doc_loader.get_doc_type_name(doc.ctype)}] {doc.title}")
        print(f"     ID: {doc.doc_id}")
        print(f"     创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")
        print()
else:
    print("我没有创建任何文档")

# ============ 我最新创建的文档 (问题 31-50) ============
print("\n【问题 31-50】我最新/最早创建的文档")

if created_docs:
    # 按创建时间排序
    sorted_docs = sorted(created_docs, key=lambda x: x.create_time)

    # 最早创建
    earliest = sorted_docs[0]
    print(f"\n最早创建的文档:")
    print(f"  标题: {earliest.title}")
    print(f"  创建时间: {doc_loader.timestamp_to_datetime(earliest.create_time)}")

    # 最新创建
    latest = sorted_docs[-1]
    print(f"\n最新创建的文档:")
    print(f"  标题: {latest.title}")
    print(f"  创建时间: {doc_loader.timestamp_to_datetime(latest.create_time)}")
else:
    print("\n没有创建过文档，无法查询最早/最新创建时间")

# ============ 文档统计 ============
print("\n=== 文档统计 ===")

stats = doc_loader.get_doc_stats(CURRENT_USER_ID)
print(f"\n可访问文档总数: {stats['accessible_count']}")
print(f"我创建的文档数: {stats['created_count']}")

print("\n按类型统计:")
for ctype, count in stats['type_counts'].items():
    type_name = doc_loader.get_doc_type_name(ctype)
    print(f"  {type_name}: {count}个")

# ============ 所有文档列表（按创建时间排序） ============
print("\n=== 所有可访问文档（按创建时间排序）===")

if accessible_docs:
    sorted_by_time = sorted(accessible_docs, key=lambda x: x.create_time)

    print("\n从最早到最新:")
    for i, doc in enumerate(sorted_by_time, 1):
        time_str = doc_loader.timestamp_to_datetime(doc.create_time)
        print(f"  {i}. [{time_str}] {doc.title}")

    print("\n从最新到最早:")
    for i, doc in enumerate(reversed(sorted_by_time), 1):
        time_str = doc_loader.timestamp_to_datetime(doc.create_time)
        print(f"  {i}. [{time_str}] {doc.title}")

print("\n" + "=" * 60)
```
