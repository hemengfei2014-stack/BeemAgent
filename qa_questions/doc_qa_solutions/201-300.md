# 问题 201-300: 更多关键词搜索、共享文档、特定文档详情

## 问题描述分类
- 问题 201-270: 更多关键词搜索（挑战赛、封号、申诉、推荐、运营、北京、上海）
- 问题 271-280: 别人共享给我的文档
- 问题 281-290: 我自己创建的文档
- 问题 291-300: 特定文档详情

## 代码实现

```python
from utils import doc_loader, CURRENT_USER_ID

print("=" * 60)
print("文档系统查询 - 更多搜索、共享文档、文档详情")
print("=" * 60)

# ============ 更多关键词搜索 (问题 201-270) ============
print("\n=== 更多关键词搜索 ===")

keywords = {
    "挑战赛": ["挑战赛"],
    "封号": ["封号"],
    "申诉": ["申诉"],
    "推荐": ["推荐"],
    "运营": ["运营"],
    "北京": ["北京"],
    "上海": ["上海"],
}

for category, kw_list in keywords.items():
    print(f"\n【{category}】")

    results = []
    for kw in kw_list:
        docs = doc_loader.search_docs_by_keyword(kw, CURRENT_USER_ID)
        for doc in docs:
            if doc.doc_id not in [d.doc_id for d in results]:
                results.append(doc)

    if results:
        print(f"找到 {len(results)} 个文档:")
        for doc in results:
            print(f"  - {doc.title} ({doc_loader.get_doc_type_name(doc.ctype)})")
    else:
        print("未找到相关文档")

# ============ 共享文档 (问题 271-280) ============
print("\n=== 别人共享给我的文档 ===")

accessible_docs = doc_loader.get_user_docs(CURRENT_USER_ID)
created_docs = doc_loader.get_user_created_docs(CURRENT_USER_ID)

# 找出不是自己创建的文档
shared_docs = [doc for doc in accessible_docs if doc.owner != CURRENT_USER_ID]

print(f"\n别人共享给我的文档: {len(shared_docs)}个")
for doc in shared_docs:
    print(f"  - {doc.title}")
    print(f"    创建者: {doc_loader.get_user_name(doc.owner)}")
    print(f"    类型: {doc_loader.get_doc_type_name(doc.ctype)}")
    print(f"    创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")

# ============ 我自己创建的文档 (问题 281-290) ============
print("\n=== 我自己创建的文档 ===")

print(f"\n我创建的文档: {len(created_docs)}个")
if created_docs:
    for doc in created_docs:
        print(f"  - {doc.title}")
        print(f"    类型: {doc_loader.get_doc_type_name(doc.ctype)}")
        print(f"    创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")
else:
    print("  没有创建任何文档")

# ============ 特定文档详情 (问题 291-300) ============
print("\n=== 特定文档详情 ===")

# 搜索"推荐相关"文档
target_docs = doc_loader.search_docs_by_keyword("推荐", CURRENT_USER_ID)

print(f"\n标题含'推荐'的文档: {len(target_docs)}个")
for doc in target_docs:
    print(f"\n文档标题: {doc.title}")
    print(f"文档ID: {doc.doc_id}")
    print(f"类型: {doc_loader.get_doc_type_name(doc.ctype)}")
    print(f"创建者: {doc_loader.get_user_name(doc.owner)}")
    print(f"创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")
    print(f"内容: {doc.note[:200]}...")

print("\n" + "=" * 60)
```
