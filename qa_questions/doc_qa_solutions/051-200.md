# 问题 51-200: 最近访问/类型统计/关键词搜索

## 问题描述分类
- 问题 51-60: 最近访问/修改的文档（数据中无此信息）
- 问题 61-110: 按类型统计文档数量
- 问题 111-200: 关键词搜索文档

## 代码实现

```python
from utils import doc_loader, CURRENT_USER_ID

print("=" * 60)
print("文档系统查询 - 类型统计、关键词搜索")
print("=" * 60)

# ============ 最近访问/修改的文档 (问题 51-60) ============
print("\n【问题 51-60】最近访问/修改的文档")
print("数据表中没有记录文档的访问/修改历史，无法提供此信息")

# ============ 按类型统计文档数量 (问题 61-110) ============
print("\n=== 按类型统计文档数量 ===")

accessible_docs = doc_loader.get_user_docs(CURRENT_USER_ID)

# 按类型统计
type_counts = {}
for doc in accessible_docs:
    ctype = doc.ctype
    type_counts[ctype] = type_counts.get(ctype, 0) + 1

print("\n文档类型统计:")
for ctype in ['doc', 'sheet', 'slide', 'pdf']:
    count = type_counts.get(ctype, 0)
    type_name = doc_loader.get_doc_type_name(ctype)
    print(f"  {type_name} ({ctype}): {count}个")

# 列出每种类型的文档
print("\n=== 各类型文档详细列表 ===")

for ctype in ['doc', 'sheet', 'slide', 'pdf']:
    print(f"\n【{doc_loader.get_doc_type_name(ctype)}】")
    docs = doc_loader.get_docs_by_type(ctype, CURRENT_USER_ID)

    if docs:
        print(f"数量: {len(docs)}个")
        for i, doc in enumerate(docs, 1):
            print(f"  {i}. {doc.title}")
            print(f"     创建者: {doc_loader.get_user_name(doc.owner)}")
            print(f"     创建时间: {doc_loader.timestamp_to_datetime(doc.create_time)}")
    else:
        print("数量: 0个")

# ============ 关键词搜索 (问题 111-200) ============
print("\n=== 关键词搜索 ===")

# 定义所有搜索关键词
keywords = {
    "MCN": ["MCN"],
    "网红": ["网红"],
    "特效": ["特效"],
    "带货": ["带货"],
    "剪辑": ["剪辑"],
    "推荐": ["推荐"],
    "算法": ["算法"],
    "流量": ["流量"],
    "审核": ["审核"],
    "申诉": ["申诉"],
    "挑战赛": ["挑战赛"],
    "粉丝": ["粉丝"],
    "封号": ["封号"],
    "运营": ["运营"],
}

for category, kw_list in keywords.items():
    print(f"\n【{category}相关文档】")

    results = []
    for kw in kw_list:
        docs = doc_loader.search_docs_by_keyword(kw, CURRENT_USER_ID)
        for doc in docs:
            if doc.doc_id not in [d.doc_id for d in results]:
                results.append(doc)

    if results:
        print(f"找到 {len(results)} 个文档:")
        for i, doc in enumerate(results, 1):
            print(f"  {i}. {doc.title}")
            print(f"     类型: {doc_loader.get_doc_type_name(doc.ctype)}")
            print(f"     创建者: {doc_loader.get_user_name(doc.owner)}")
    else:
        print("未找到相关文档")

# ============ 按创建者统计 ============
print("\n=== 按创建者统计 ===")

creator_counts = {}
for doc in accessible_docs:
    creator = doc_loader.get_user_name(doc.owner)
    creator_counts[creator] = creator_counts.get(creator, 0) + 1

print("\n各创建者的文档数:")
for creator, count in sorted(creator_counts.items(), key=lambda x: x[1], reverse=True):
    print(f"  {creator}: {count}个")

# ============ 文档时间范围 ============
print("\n=== 文档时间范围 ===")

if accessible_docs:
    times = [doc.create_time for doc in accessible_docs]
    earliest = min(times)
    latest = max(times)

    print(f"最早创建: {doc_loader.timestamp_to_datetime(earliest)}")
    print(f"最新创建: {doc_loader.timestamp_to_datetime(latest)}")

print("\n" + "=" * 60)
```
