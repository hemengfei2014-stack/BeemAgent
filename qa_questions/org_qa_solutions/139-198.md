## 代码实现

```python
from utils import loader, CURRENT_ORG_ID, CURRENT_USER_ID

# 查找贝冠
name = "贝冠"
users = loader.get_user_by_name(name, CURRENT_ORG_ID)

print(f"=== 查找姓名为'{name}'的员工 ===")
print(f"找到 {len(users)} 个结果\n")

if len(users) == 0:
    print("没有找到该员工")
elif len(users) == 1:
    user = users[0]
    print(f"唯一匹配: {user.name} (工号:{user.id})")

    # 基本信息
    print("\n=== 基本信息 ===")
    print(f"工号: {user.id}")
    print(f"姓名: {user.name}")
    print(f"性别: {loader.get_gender_name(user.gender)}")
    print(f"生日: {loader.timestamp_to_date(user.birthday)}")
    print(f"角色: {loader.get_role_name(user.role)}")
    print(f"手机号: {user.mobile}")
    print(f"邮箱: {user.email}")

    # 入职信息
    print("\n=== 入职信息 ===")
    print(f"入职日期: {loader.timestamp_to_date(user.join_date)}")
    print(f"入职年份: {loader.timestamp_to_year(user.join_date)}")

    # 部门信息
    print("\n=== 部门信息 ===")
    my_depts = loader.get_user_depts(user.id)
    print(f"归属部门数量: {len(my_depts)}")
    for dept in my_depts:
        print(f"  - {dept.name} (ID:{dept.id}, Level:{dept.level})")

    # 主部门
    main_dept = loader.get_user_dept(user.id)
    if main_dept:
        print(f"\n主部门: {main_dept.name}")
        print(f"部门层级: Level {main_dept.level}")

        # 一级部门
        if main_dept.level == 1:
            print(f"一级部门: {main_dept.name}")
        else:
            # 需要找到一级父部门
            current = main_dept
            while current and current.level > 1:
                parent_id = current.parent_id
                if parent_id:
                    current = loader.get_dept_by_id(int(parent_id))
                else:
                    break
            if current:
                print(f"一级部门: {current.name}")

    # 领导信息
    print("\n=== 领导信息 ===")
    if user.manager_id:
        manager = loader.get_user_by_id(int(user.manager_id))
        if manager:
            print(f"直属领导: {manager.name} (工号:{manager.id})")

    # 下属信息
    print("\n=== 下属信息 ===")
    subs = loader.get_subordinates(user.id, CURRENT_ORG_ID)
    print(f"直属下属数量: {len(subs)}")
    if subs:
        for sub in subs:
            print(f"  - {sub.name} (工号:{sub.id})")

    # 和"我"的关系
    print("\n=== 和'我'(籍钧良)的关系 ===")
    me = loader.get_user_by_id(CURRENT_USER_ID)
    my_dept = loader.get_user_dept(CURRENT_USER_ID)
    bei_dept = loader.get_user_dept(user.id)

    if my_dept and bei_dept:
        same_dept = my_dept.id == bei_dept.id
        print(f"是否同一部门: {'是' if same_dept else '否'}")
        print(f"我的部门: {my_dept.name}")
        print(f"贝冠的部门: {bei_dept.name}")

    # 共同上级
    if me.manager_id and user.manager_id and me.manager_id == user.manager_id:
        common_manager = loader.get_user_by_id(int(me.manager_id))
        print(f"共同上级: {common_manager.name}")
    else:
        print("没有共同的直属上级")

else:
    # 多个结果
    print("多个匹配结果:")
    for user in users:
        dept = loader.get_user_dept(user.id)
        dept_name = dept.name if dept else "未分配"
        print(f"  - {user.name} (工号:{user.id}, 部门:{dept_name})")
```
