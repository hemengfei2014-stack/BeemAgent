# 问题 31-60: 部门数量、性别比例、平均人数、最大最小部门

## 问题描述
- 问题 31-36: 部门总数
- 问题 37-42: 性别比例/男女比例
- 问题 43-48: 部门平均人数
- 问题 49-54: 人数最多的部门
- 问题 55-60: 人数最少的部门

## 代码实现

```python
from utils import loader, CURRENT_ORG_ID
import pandas as pd

# ============ 部门总数 ============
print("=== 部门总数 ===")
depts = loader.get_org_depts(CURRENT_ORG_ID)
total_depts = len(depts)
print(f"Jaco组织部门总数: {total_depts} 个")

# 列出所有部门
print("\n所有部门列表:")
for dept in depts:
    dept_users = loader.get_dept_users(dept.id)
    print(f"  - {dept.name} (ID:{dept.id}): {len(dept_users)}人")

# ============ 性别比例 ============
print("\n=== 性别比例 ===")
users = loader.get_org_users(CURRENT_ORG_ID)
male_count = sum(1 for u in users if u.gender == 1)
female_count = sum(1 for u in users if u.gender == 2)
unknown_count = sum(1 for u in users if u.gender == 0)

print(f"男性员工: {male_count} 人")
print(f"女性员工: {female_count} 人")
print(f"未知性别: {unknown_count} 人")
print(f"\n性别比例: 男性 {'更多' if male_count > female_count else '较少' if male_count < female_count else '相等'}")
print(f"男女比例: {male_count}:{female_count}")

# ============ 部门平均人数 ============
print("\n=== 部门平均人数 ===")
dept_counts = [(dept.name, loader.get_dept_count(dept.id)) for dept in depts]
avg_count = sum(count for _, count in dept_counts) / len(dept_counts)
print(f"部门平均人数: {avg_count:.1f} 人")

# ============ 人数最多的部门 ============
print("\n=== 人数最多的部门 ===)
max_dept = max(dept_counts, key=lambda x: x[1])
print(f"部门名称: {max_dept[0]}")
print(f"人数: {max_dept[1]} 人")

# 按人数排序列出所有部门
sorted_depts = sorted(dept_counts, key=lambda x: x[1], reverse=True)
print("\n部门人数排序:")
for name, count in sorted_depts:
    print(f"  {name}: {count}人")

# ============ 人数最少的部门 ============
print("\n=== 人数最少的部门 ===")
min_dept = min(dept_counts, key=lambda x: x[1])
print(f"部门名称: {min_dept[0]}")
print(f"人数: {min_dept[1]} 人")
```
