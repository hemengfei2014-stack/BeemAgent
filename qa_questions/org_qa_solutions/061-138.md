## 代码实现

```python
from utils import loader, CURRENT_ORG_ID, CURRENT_USER_ID

# ============ 我的个人资料 ============
print("=== 我的个人资料 ===")
me = loader.get_user_by_id(CURRENT_USER_ID)
print(f"工号(User ID): {me.id}")
print(f"姓名: {me.name}")
print(f"组织ID: {me.org_id}")
print(f"角色ID: {me.role}")
print(f"角色名称: {loader.get_role_name(me.role)}")
print(f"性别: {loader.get_gender_name(me.gender)}")

# 入职时间
join_date = loader.timestamp_to_date(me.join_date)
join_year = loader.timestamp_to_year(me.join_date)
print(f"入职日期: {join_date}")
print(f"入职年份: {join_year}")

# 联系方式
print(f"手机号: {me.mobile}")
print(f"邮箱: {me.email}")

# 生日
birthday = loader.timestamp_to_date(me.birthday)
print(f"生日: {birthday}")

# ============ 我的上级/直属领导 ============
print("\n=== 我的直属领导 ===")
if me.manager_id:
    manager = loader.get_user_by_id(int(me.manager_id))
    if manager:
        print(f"我的上级: {manager.name}")
        print(f"上级工号: {manager.id}")
        print(f"上级职位: {loader.get_role_name(manager.role)}")
else:
    print("我没有直属领导（我是CEO）")

# ============ 我所属的部门 ============
print("\n=== 我所属的部门 ===")
my_dept = loader.get_user_dept(CURRENT_USER_ID)
if my_dept:
    print(f"部门ID: {my_dept.id}")
    print(f"部门名称: {my_dept.name}")
    print(f"部门层级: Level {my_dept.level}")

    # 部门负责人
    if my_dept.leader_id:
        leader = loader.get_user_by_id(int(my_dept.leader_id))
        if leader:
            print(f"部门负责人: {leader.name}（工号:{leader.id}）")

    # 部门路径
    path = loader.get_dept_tree_path(my_dept.id)
    print(f"部门层级路径: {path}")

# ============ 我的一级部门 ============
print("\n=== 我的一级部门归属 ===")
my_depts = loader.get_user_depts(CURRENT_USER_ID)
for dept in my_depts:
    if dept.level == 1:
        print(f"一级部门: {dept.name}")
        break

# ============ 兄弟部门（同级部门） ============
print("\n=== 兄弟部门（同级部门） ===")
siblings = loader.get_sibling_depts(my_dept.id)
print(f"兄弟部门数量: {len(siblings)}")
for sib in siblings:
    count = loader.get_dept_count(sib.id)
    print(f"  - {sib.name}: {count}人")

# ============ 我的下属 ============
print("\n=== 我的下属 ===")
subordinates = loader.get_subordinates(CURRENT_USER_ID, CURRENT_ORG_ID)
print(f"直属下属数量: {len(subordinates)}")
for sub in subordinates:
    sub_dept = loader.get_user_dept(sub.id)
    dept_name = sub_dept.name if sub_dept else "未分配部门"
    print(f"  - {sub.name} (工号:{sub.id}, 角色:{loader.get_role_name(sub.role)}, 部门:{dept_name})")
```
