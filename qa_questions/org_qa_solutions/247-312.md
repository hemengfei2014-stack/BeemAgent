# 问题 247-312: 特定部门查询

## 问题描述
- 问题 247-252: 财务部详细信息
- 问题 253-258: 研发部负责人
- 问题 259-264: 产品部人数
- 问题 265-270: 市场部上级部门
- 问题 271-276: 研发部子部门
- 问题 277-282: 人力部门人数
- 问题 283-288: 用增团队负责人
- 问题 289-294: 基架部门上级
- 问题 295-300: 产研中心人数
- 问题 301-306: 搜索含"效能"的部门
- 问题 307-312: 搜索含"张"的员工

## 代码实现

```python
from utils import loader, CURRENT_ORG_ID

print("=" * 60)
print("特定部门查询")
print("=" * 60)

# ============ 财务部 ============
print("\n【问题 247-252】财务部查询")
depts = loader.get_dept_by_name("财务", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
        user_count = loader.get_dept_count(dept.id)
        print(f"部门名称: {dept.name}")
        print(f"部门ID: {dept.id}")
        print(f"负责人: {leader.name if leader else '无'}")
        print(f"人数: {user_count}")
else:
    print("Jaco组织没有'财务部'")

# ============ 研发部 ============
print("\n【问题 253-258】研发部查询")
depts = loader.get_dept_by_name("研发", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
        print(f"部门名称: {dept.name}")
        print(f"负责人: {leader.name if leader else '无'}")

        # 子部门
        children = loader.get_child_depts(dept.id)
        if children:
            print(f"子部门: {[c.name for c in children]}")
        else:
            print("子部门: 无")
else:
    print("Jaco组织没有'研发部'")
    print("可能相关: 后端部 (负责人:籍钧良)")

# ============ 产品部 ============
print("\n【问题 259-264】产品部查询")
depts = loader.get_dept_by_name("产品", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        user_count = loader.get_dept_count(dept.id)
        print(f"部门名称: {dept.name}")
        print(f"人数: {user_count}人")

        # 成员列表
        users = loader.get_dept_users(dept.id)
        print(f"成员: {', '.join([u.name for u in users])}")
else:
    print("没有找到产品部")

# ============ 市场部 ============
print("\n【问题 265-270】市场部查询")
depts = loader.get_dept_by_name("市场", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        parent_id = dept.parent_id
        if parent_id:
            parent = loader.get_dept_by_id(int(parent_id))
            print(f"部门名称: {dept.name}")
            print(f"上级部门: {parent.name if parent else '无'}")
        else:
            print(f"部门名称: {dept.name} 是一级部门，无上级部门")
else:
    print("Jaco组织没有'市场部'")

# ============ 人力部门 ============
print("\n【问题 277-282】人力部门查询")
depts = loader.get_dept_by_name("人力", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        user_count = loader.get_dept_count(dept.id)
        print(f"部门名称: {dept.name}")
        print(f"人数: {user_count}人")
else:
    # 尝试查找"人力资源"
    depts = loader.get_dept_by_name("人力资源", CURRENT_ORG_ID)
    if depts:
        for dept in depts:
            user_count = loader.get_dept_count(dept.id)
            print(f"部门名称: {dept.name}")
            print(f"人数: {user_count}人")

# ============ 用增（用户增长） ============
print("\n【问题 283-288】用增团队查询")
depts = loader.get_dept_by_name("用增", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
        print(f"部门名称: {dept.name}")
        print(f"负责人: {leader.name if leader else '无'}")
else:
    # 尝试"用户增长"
    depts = loader.get_dept_by_name("用户增长", CURRENT_ORG_ID)
    if depts:
        for dept in depts:
            leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
            print(f"部门名称: {dept.name} (简称:用增)")
            print(f"负责人: {leader.name if leader else '无'}")

# ============ 基架（基础架构） ============
print("\n【问题 289-294】基架部门查询")
depts = loader.get_dept_by_name("基架", CURRENT_ORG_ID)
if depts:
    for dept in depts:
        parent_id = dept.parent_id
        if parent_id:
            parent = loader.get_dept_by_id(int(parent_id))
            print(f"部门名称: {dept.name}")
            print(f"上级部门: {parent.name if parent else '无'}")
        else:
            print(f"部门名称: {dept.name} 是一级部门")
else:
    # 尝试"基础架构"
    depts = loader.get_dept_by_name("基础架构", CURRENT_ORG_ID)
    if depts:
        for dept in depts:
            parent_id = dept.parent_id
            if parent_id:
                parent = loader.get_dept_by_id(int(parent_id))
                print(f"部门名称: {dept.name}")
                print(f"上级部门: {parent.name if parent else '无'}")
    else:
        print("Jaco组织没有'基架'或'基础架构'部门")

# ============ 产研中心 ============
print("\n【问题 295-300】产研中心查询")
# 产研可能指产品+研发
depts = loader.get_dept_by_name("产品", CURRENT_ORG_ID)
product_count = 0
if depts:
    for dept in depts:
        product_count += loader.get_dept_count(dept.id)

# 研发相关：后端、前端、机器学习
rd_depts = loader.get_dept_by_name("后端", CURRENT_ORG_ID) + \
           loader.get_dept_by_name("前端", CURRENT_ORG_ID) + \
           loader.get_dept_by_name("机器学习", CURRENT_ORG_ID)
rd_count = sum(loader.get_dept_count(d.id) for d in rd_depts)

total = product_count + rd_count
print(f"产品部人数: {product_count}")
print(f"研发相关人数（后端+前端+机器学习）: {rd_count}")
print(f"产研中心总人数: {total}")

# ============ 效能部门 ============
print("\n【问题 301-306】效能部门查询")
all_depts = loader.get_org_depts(CURRENT_ORG_ID)
efficiency_depts = [d for d in all_depts if "效能" in d.name]
if efficiency_depts:
    for dept in efficiency_depts:
        print(f"部门名称: {dept.name}")
else:
    print("Jaco组织没有名称含'效能'的部门")

# ============ 姓张的员工 ============
print("\n【问题 307-312】姓张的员工查询")
users = loader.get_users_by_name_pattern("张", CURRENT_ORG_ID)
print(f"找到 {len(users)} 位姓张的员工:")
for user in users:
    dept = loader.get_user_dept(user.id)
    dept_name = dept.name if dept else "未分配"
    print(f"  - {user.name} (工号:{user.id}, 部门:{dept_name})")
```
