# 问题 229-246: 部门列表和组织架构

## 问题描述
- 问题 229-234: 列出所有部门名称
- 问题 235-240: 列出所有一级部门
- 问题 241-246: 组织架构树

## 代码实现

```python
from utils import loader, CURRENT_ORG_ID

print("=== Jaco组织所有部门 ===\n")

# 获取所有部门
all_depts = loader.get_org_depts(CURRENT_ORG_ID)
print(f"部门总数: {len(all_depts)}\n")

print("所有部门列表:")
for dept in all_depts:
    user_count = loader.get_dept_count(dept.id)
    leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
    leader_name = leader.name if leader else "无"
    print(f"  [{dept.id}] {dept.name} - {dept.level}级部门 - {user_count}人 - 负责人:{leader_name}")

# ============ 一级部门 ============
print("\n=== 一级部门 ===\n")
top_depts = loader.get_top_level_depts(CURRENT_ORG_ID)
print(f"一级部门数量: {len(top_depts)}\n")

for dept in top_depts:
    user_count = loader.get_dept_count(dept.id)
    leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
    leader_name = leader.name if leader else "无"
    print(f"  [{dept.id}] {dept.name} - {user_count}人 - 负责人:{leader_name}")

# ============ 组织架构树 ============
print("\n=== 组织架构树 ===\n")
print("Jaco (组织)")
print("├── 一级部门结构:")

for dept in top_depts:
    user_count = loader.get_dept_count(dept.id)
    leader = loader.get_user_by_id(int(dept.leader_id)) if dept.leader_id else None
    leader_name = leader.name if leader else "无"
    print(f"│   ├── {dept.name} (负责人:{leader_name}, {user_count}人)")

    # 查找子部门
    children = loader.get_child_depts(dept.id)
    if children:
        for child in children:
            child_count = loader.get_dept_count(child.id)
            child_leader = loader.get_user_by_id(int(child.leader_id)) if child.leader_id else None
            child_leader_name = child_leader.name if child_leader else "无"
            print(f"│   │   └── {child.name} (负责人:{child_leader_name}, {child_count}人)")

# ============ 详细部门信息 ============
print("\n=== 详细部门信息 ===\n")
for dept in all_depts:
    print(f"部门ID: {dept.id}")
    print(f"部门名称: {dept.name}")
    print(f"组织ID: {dept.org_id}")
    print(f"父部门ID: {dept.parent_id if dept.parent_id else '无（一级部门）'}")
    print(f"部门层级: Level {dept.level}")
    print(f"负责人ID: {dept.leader_id}")

    # 获取负责人信息
    if dept.leader_id:
        leader = loader.get_user_by_id(int(dept.leader_id))
        if leader:
            print(f"负责人姓名: {leader.name}")
            print(f"负责人角色: {loader.get_role_name(leader.role)}")

    # 获取部门成员
    users = loader.get_dept_users(dept.id)
    print(f"部门成员数: {len(users)}")
    print(f"成员名单: {', '.join([u.name for u in users])}")
    print("-" * 50)
```
