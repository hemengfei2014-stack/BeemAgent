## 代码实现

```python
from utils import loader, CURRENT_ORG_ID
from datetime import datetime

print("=" * 60)
print("员工搜索和角色统计")
print("=" * 60)

# ============ 手机号尾号8888 ============
print("\n【问题 313-318】手机号尾号8888的员工")
users = loader.get_user_by_mobile("8888", CURRENT_ORG_ID)
if users:
    for user in users:
        dept = loader.get_user_dept(user.id)
        dept_name = dept.name if dept else "未分配"
        print(f"姓名: {user.name}")
        print(f"工号: {user.id}")
        print(f"手机号: {user.mobile}")
        print(f"部门: {dept_name}")
else:
    print("没有找到手机号尾号8888的员工")

# ============ 邮箱查询 ============
print("\n【问题 319-324】邮箱查询")
email = "kai.bei@company.com"
user = loader.get_user_by_email(email, CURRENT_ORG_ID)
if user:
    print(f"姓名: {user.name}")
    print(f"工号: {user.id}")
    print(f"邮箱: {user.email}")
else:
    print(f"没有找到邮箱为 {email} 的员工")

# ============ 产品经理数量 ============
print("\n【问题 325-330】产品经理统计")
ROLE_PRODUCT_MANAGER = 1
pms = loader.get_users_by_role(ROLE_PRODUCT_MANAGER, CURRENT_ORG_ID)
print(f"产品经理数量: {len(pms)}")
print("产品经理列表:")
for pm in pms:
    dept = loader.get_user_dept(pm.id)
    dept_name = dept.name if dept else "未分配"
    print(f"  - {pm.name} (工号:{pm.id}, 部门:{dept_name})")

# ============ 开发工程师列表 ============
print("\n【问题 331-336】开发工程师列表")
ROLE_DEV_ENGINEER = 2
devs = loader.get_users_by_role(ROLE_DEV_ENGINEER, CURRENT_ORG_ID)
print(f"开发工程师数量: {len(devs)}")
print("开发工程师列表:")
for dev in devs:
    dept = loader.get_user_dept(dev.id)
    dept_name = dept.name if dept else "未分配"
    print(f"  - {dev.name} (工号:{dev.id}, 部门:{dept_name})")

# ============ 研发部Android工程师 ============
print("\n【问题 337-342】研发部Android工程师")
ROLE_ANDROID = 5
# 先找研发相关部门（后端、前端、机器学习）
rd_dept_names = ["后端", "前端", "机器学习"]
android_devs = []

for dept_name in rd_dept_names:
    depts = loader.get_dept_by_name(dept_name, CURRENT_ORG_ID)
    for dept in depts:
        users = loader.get_dept_users(dept.id)
        android_in_dept = [u for u in users if u.role == ROLE_ANDROID]
        android_devs.extend(android_in_dept)

print(f"研发相关部门Android工程师数量: {len(android_devs)}")
for dev in android_devs:
    dept = loader.get_user_dept(dev.id)
    print(f"  - {dev.name} (工号:{dev.id}, 部门:{dept.name if dept else '未分配'})")

if len(android_devs) == 0:
    print("Jaco组织研发相关部门没有Android工程师")

# ============ 测试工程师数量 ============
print("\n【问题 343-348】测试工程师统计")
ROLE_TEST_ENGINEER = 7
testers = loader.get_users_by_role(ROLE_TEST_ENGINEER, CURRENT_ORG_ID)
print(f"测试工程师数量: {len(testers)}")
if len(testers) > 0:
    for tester in testers:
        dept = loader.get_user_dept(tester.id)
        dept_name = dept.name if dept else "未分配"
        print(f"  - {tester.name} (工号:{tester.id}, 部门:{dept_name})")
else:
    print("Jaco组织没有测试工程师")

# ============ HR列表 ============
print("\n【问题 349-355】HR列表")
ROLE_HR = 4
hrs = loader.get_users_by_role(ROLE_HR, CURRENT_ORG_ID)
print(f"HR数量: {len(hrs)}")
print("HR列表:")
for hr in hrs:
    dept = loader.get_user_dept(hr.id)
    dept_name = dept.name if dept else "未分配"
    print(f"  - {hr.name} (工号:{hr.id}, 部门:{dept_name})")

# 招聘专员 - 这里假设HR部门中负责招聘的HR
# 由于数据中没有细分招聘职能，列出所有HR
print("\n【问题 356-360】招聘专员")
if len(hrs) > 0:
    print("HR团队成员（可能负责招聘）:")
    for hr in hrs:
        print(f"  - {hr.name}")

# ============ 一级部门leader ============
print("\n【问题 361-366】一级部门leader")
ROLE_LEVEL1_LEADER = 9
level1_leaders = loader.get_users_by_role(ROLE_LEVEL1_LEADER, CURRENT_ORG_ID)
print(f"一级部门leader数量: {len(level1_leaders)}")
print("一级部门leader列表:")
for leader in level1_leaders:
    # 查找负责的部门
    depts = loader.get_org_depts(CURRENT_ORG_ID)
    led_depts = [d for d in depts if d.leader_id == leader.id]
    dept_names = ", ".join([d.name for d in led_depts])
    print(f"  - {leader.name} (工号:{leader.id}, 负责部门:{dept_names if dept_names else '无'})")

# ============ 二级部门leader ============
print("\n【问题 367-372】二级部门leader")
ROLE_LEVEL2_LEADER = 8
level2_leaders = loader.get_users_by_role(ROLE_LEVEL2_LEADER, CURRENT_ORG_ID)
print(f"二级部门leader数量: {len(level2_leaders)}")
if len(level2_leaders) > 0:
    for leader in level2_leaders:
        depts = loader.get_org_depts(CURRENT_ORG_ID)
        led_depts = [d for d in depts if d.leader_id == leader.id]
        dept_names = ", ".join([d.name for d in led_depts])
        print(f"  - {leader.name} (工号:{leader.id}, 负责部门:{dept_names if dept_names else '无'})")
else:
    print("Jaco组织没有二级部门leader")

# ============ 入职时间统计 ============
print("\n【问题 373-390】入职时间统计")
users = loader.get_org_users(CURRENT_ORG_ID)

# 按入职时间排序
sorted_users = sorted(users, key=lambda u: u.join_date)

# 最早入职
earliest = sorted_users[0]
earliest_dept = loader.get_user_dept(earliest.id)
print(f"\n最早入职员工:")
print(f"  姓名: {earliest.name}")
print(f"  工号: {earliest.id}")
print(f"  入职日期: {loader.timestamp_to_date(earliest.join_date)}")
print(f"  部门: {earliest_dept.name if earliest_dept else '未分配'}")

# 最近入职
latest = sorted_users[-1]
latest_dept = loader.get_user_dept(latest.id)
print(f"\n最近入职员工:")
print(f"  姓名: {latest.name}")
print(f"  工号: {latest.id}")
print(f"  入职日期: {loader.timestamp_to_date(latest.join_date)}")
print(f"  部门: {latest_dept.name if latest_dept else '未分配'}")

# 今年入职
current_year = datetime.now().year
this_year_users = [u for u in users if loader.timestamp_to_year(u.join_date) == current_year]
print(f"\n今年({current_year}年)入职员工:")
print(f"  数量: {len(this_year_users)}")
if len(this_year_users) > 0:
    for user in this_year_users:
        print(f"  - {user.name} (工号:{user.id}, 入职日期:{loader.timestamp_to_date(user.join_date)})")
else:
    print(f"  今年没有新入职员工（所有员工都是{loader.timestamp_to_year(earliest.join_date)}年入职）")
```
