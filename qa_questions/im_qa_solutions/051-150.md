def count_messages_days(user_id, days=7):
    """统计指定天数内与某用户的消息数"""
    cutoff_time = datetime.now().timestamp() * 1000 - (days * 24 * 3600 * 1000)

    messages = im_loader.get_messages_between_users(CURRENT_USER_ID, user_id)
    recent = messages[messages['send_time_ms'] >= cutoff_time]

    return len(recent)
