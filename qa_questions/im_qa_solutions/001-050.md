# 问题 1-50: 最近联系人、发送/接收时间、消息统计

## 问题描述
- 问题 1-6: 最近的沟通对象/聊天联系人
- 问题 7-12: 最近一次发送消息时间
- 问题 13-18: 最近一次接收消息时间
- 问题 19-24: 谁给我发消息最多
- 问题 24-28: 最近联系最频繁的人
- 问题 29-40: 今天发送/接收消息数量
- 问题 41-50: 最近的会话列表和未回复会话

## 代码实现

```python
from utils import im_loader, CURRENT_USER_ID
from datetime import datetime, date

print("=" * 60)
print(f"IM系统查询 - 当前用户: {im_loader.get_user_name(CURRENT_USER_ID)} (ID:{CURRENT_USER_ID})")
print("=" * 60)

# ============ 最近的沟通对象 (问题 1-6) ============
print("\n【问题 1-6】最近的沟通对象")
recent_contacts = im_loader.get_recent_contacts(CURRENT_USER_ID, limit=20)
print(f"最近联系的联系人数量: {len(recent_contacts)}")

for i, contact in enumerate(recent_contacts, 1):
    if contact['is_group']:
        print(f"  {i}. 群聊: {contact['id']}")
    else:
        print(f"  {i}. {im_loader.get_user_name(contact['id'])} (ID:{contact['id']})")
    print(f"     最后联系时间: {im_loader.timestamp_to_datetime(contact['last_time'])}")

# ============ 最近一次发送/接收消息时间 (问题 7-18) ============
print("\n【问题 7-12】最近一次发送消息时间")
sent = im_loader.get_user_sent_messages(CURRENT_USER_ID)
if len(sent) > 0:
    latest_sent = sent.sort_values('send_time_ms', ascending=False).iloc[0]
    print(f"最近一次发送: {im_loader.timestamp_to_datetime(latest_sent['send_time_ms'])}")
    print(f"会话ID: {latest_sent['session_id']}")
    print(f"消息类型: {im_loader.MESSAGE_TYPE.get(latest_sent['message_type'], '未知')}")
else:
    print("没有发送过消息")

print("\n【问题 13-18】最近一次接收消息时间")
received = im_loader.get_user_received_messages(CURRENT_USER_ID)
if len(received) > 0:
    latest_received = received.sort_values('send_time_ms', ascending=False).iloc[0]
    print(f"最近一次接收: {im_loader.timestamp_to_datetime(latest_received['send_time_ms'])}")
    print(f"发送者: {im_loader.get_user_name(latest_received['from_user_id'])}")
    print(f"会话ID: {latest_received['session_id']}")
else:
    print("没有收到过消息")

# ============ 谁给我发消息最多 (问题 19-24) ============
print("\n【问题 19-24】发给我消息最多的Top 5")
sender_counts = im_loader.get_message_count_by_sender(CURRENT_USER_ID)
sorted_senders = sorted(sender_counts.items(), key=lambda x: x[1], reverse=True)[:5]

print("发消息最多的人 (Top 5):")
for i, (sender_id, count) in enumerate(sorted_senders, 1):
    print(f"  {i}. {im_loader.get_user_name(sender_id)} - {count}条消息")

# ============ 今天发送/接收消息统计 (问题 29-40) ============
print("\n【问题 29-40】今天消息统计")
today = date.today().strftime('%Y-%m-%d')
stats = im_loader.get_message_stats(CURRENT_USER_ID, date=today)

print(f"日期: {today}")
print(f"发送总数: {stats['sent_total']}条")
print(f"  - 私聊: {stats['sent_private']}条")
print(f"  - 群聊: {stats['sent_group']}条")
print(f"接收总数: {stats['received_total']}条")
print(f"  - 私聊: {stats['received_private']}条")
print(f"  - 群聊: {stats['received_group']}条")

# 按小时统计今天的发送量
print("\n按小时统计发送量:")
sent_today = im_loader.get_user_sent_messages(CURRENT_USER_ID)
start_ts = int(datetime.combine(date.today(), datetime.min.time()).timestamp() * 1000)
sent_today = sent_today[sent_today['send_time_ms'] >= start_ts]

hourly_counts = {}
for _, msg in sent_today.iterrows():
    dt = datetime.fromtimestamp(msg['send_time_ms'] / 1000)
    hour = dt.hour
    hourly_counts[hour] = hourly_counts.get(hour, 0) + 1

for hour in sorted(hourly_counts.keys()):
    print(f"  {hour:02d}:00 - {hourly_counts[hour]}条")

# ============ 最近的会话列表 (问题 41-46) ============
print("\n【问题 41-46】最近5个会话")
recent_sessions = im_loader.get_recent_sessions(CURRENT_USER_ID, limit=5)

for i, session in enumerate(recent_sessions, 1):
    conv_type = im_loader.CONVERSATION_TYPE.get(session['conversation_type'], '未知')
    print(f"  {i}. 会话ID: {session['session_id']}")
    print(f"     类型: {conv_type}")
    print(f"     最后消息时间: {im_loader.timestamp_to_datetime(session['last_time'])}")
    print(f"     最后消息来自: {im_loader.get_user_name(session['last_message_from'])}")

# ============ 未回复的会话 (问题 47-50) ============
print("\n【问题 47-50】未回复的会话")
unreplied = im_loader.get_unreplied_sessions(CURRENT_USER_ID)

print(f"未回复会话数: {len(unreplied)}")
for i, session in enumerate(unreplied, 1):
    conv_type = im_loader.CONVERSATION_TYPE.get(session['conversation_type'], '未知')
    print(f"  {i}. 会话ID: {session['session_id']} ({conv_type})")
    print(f"     最后消息来自: {im_loader.get_user_name(session['last_message_from'])}")
    print(f"     时间: {im_loader.timestamp_to_datetime(session['last_time'])}")

print("\n" + "=" * 60)
```
