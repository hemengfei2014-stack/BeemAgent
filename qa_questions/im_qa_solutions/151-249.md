- 问题 150-154: 共同群聊查询
- 问题 155-202: 全局关键词搜索（短视频、算法、BGM、剪辑、带货、涨粉、竞品、城市、活动策划、预算）
- 问题 203-214: 活动策划和预算相关搜索
- 问题 215-227: 3天内附件、数据报表搜索
- 问题 227-249: 群聊列表、群聊消息统计、群成员活跃度

## 代码实现

```python
from utils import im_loader, CURRENT_USER_ID
import pandas as pd

print("=" * 60)
print("IM系统查询 - 共同群聊、全局搜索、群聊统计")
print("=" * 60)

# ============ 共同群聊查询 (问题 150-154) ============
print("\n=== 共同群聊查询 ===")

def find_common_groups(user1_id, user2_id):
    """查找两个用户的共同群聊"""
    user1_sessions = im_loader.get_user_sessions(user1_id)
    user2_sessions = im_loader.get_user_sessions(user2_id)

    user1_groups = {s[0] for s in user1_sessions if s[1] == 3}
    user2_groups = {s[0] for s in user2_sessions if s[1] == 3}

    common = user1_groups & user2_groups
    return list(common)

users_to_check = [
    (10051, "萧翔"),
    (10008, "李世"),
    (10044, "贝冠"),
]

for user_id, user_name in users_to_check:
    common_groups = find_common_groups(CURRENT_USER_ID, user_id)
    if common_groups:
        print(f"\n与{user_name}的共同群聊: {common_groups}")
    else:
        print(f"\n与{user_name}没有共同群聊")

# ============ 全局关键词搜索 (问题 155-202) ============
print("\n=== 全局关键词搜索 ===")

# 定义所有要搜索的关键词组
keyword_groups = {
    "短视频": ["短视频"],
    "算法": ["算法"],
    "BGM/音乐": ["BGM", "音乐"],
    "剪辑": ["剪辑"],
    "带货/电商": ["带货", "电商"],
    "涨粉/增长": ["涨粉", "用户增长", "增长"],
    "竞品/资料": ["竞品", "资料", "参考"],
    "城市": ["上海", "北京", "深圳", "广州"],
    "活动策划": ["活动策划", "活动方案"],
    "预算/成本": ["预算", "成本", "费用", "花费"],
}

def search_all_keywords(keywords):
    """搜索多个关键词"""
    my_sessions = im_loader.get_user_sessions(CURRENT_USER_ID)

    all_messages = []
    for session_id, conv_type in my_sessions:
        messages = im_loader.get_session_messages(session_id)

        for kw in keywords:
            matching = messages[messages['content'].str.contains(kw, na=False, case=False)]
            if not matching.empty:
                matching = matching.copy()
                matching['keyword'] = kw
                all_messages.append(matching)

    if all_messages:
        return pd.concat(all_messages)
    return pd.DataFrame()

for category, keywords in keyword_groups.items():
    results = search_all_keywords(keywords)

    if not results.empty:
        print(f"\n【{category}】找到 {len(results)} 条消息")

        # 按发送人统计
        sender_counts = results.groupby('from_user_id').size().sort_values(ascending=False)
        print("按发送人统计:")
        for sender_id, count in sender_counts.head(5).items():
            sender_name = im_loader.get_user_name(sender_id)
            print(f"  {sender_name}: {count}条")

        # 最近3条
        print("最近3条:")
        for _, msg in results.sort_values('send_time_ms', ascending=False).head(3).iterrows():
            sender = im_loader.get_user_name(msg['from_user_id'])
            time = im_loader.timestamp_to_datetime(msg['send_time_ms'])
            content = msg['content'][:50] + '...' if len(msg['content']) > 50 else msg['content']
            print(f"  [{time}] {sender}: {content}")
    else:
        print(f"\n【{category}】没有找到相关消息")

# ============ 3天内附件统计 (问题 215-220) ============
print("\n=== 3天内附件统计 ===")

from datetime import datetime, timedelta

cutoff_time = int((datetime.now() - timedelta(days=3)).timestamp() * 1000)

received = im_loader.get_user_received_messages(CURRENT_USER_ID)
recent_received = received[received['send_time_ms'] >= cutoff_time]

# 文件消息
files = recent_received[recent_received['message_type'] == 3]

print(f"\n近3天接收的文件: {len(files)}个")
for _, msg in files.sort_values('send_time_ms', ascending=False).iterrows():
    sender = im_loader.get_user_name(msg['from_user_id'])
    time = im_loader.timestamp_to_datetime(msg['send_time_ms'])
    print(f"  - {msg['content']} (来自:{sender}, 时间:{time})")

# ============ 数据报表搜索 (问题 221-226) ============
print("\n=== 数据报表搜索 ===")

report_keywords = ["数据报表", "报表", "数据统计"]
for kw in report_keywords:
    received_files = received[received['message_type'] == 3]
    matching = received_files[received_files['content'].str.contains(kw, na=False, case=False)]

    if not matching.empty:
        print(f"\n关键词'{kw}': {len(matching)}个文件")
        for _, msg in matching.iterrows():
            sender = im_loader.get_user_name(msg['from_user_id'])
            time = im_loader.timestamp_to_datetime(msg['send_time_ms'])
            print(f"  - {msg['content']} (来自:{sender}, 时间:{time})")

# ============ 我参与的群聊列表 (问题 227-232) ============
print("\n=== 我参与的群聊 ===")

my_sessions = im_loader.get_user_sessions(CURRENT_USER_ID)
group_sessions = [(sid, ctype) for sid, ctype in my_sessions if ctype == 3]

print(f"\n我参与的群聊数量: {len(group_sessions)}")

if group_sessions:
    print("群聊列表:")
    for session_id, _ in group_sessions:
        # 获取该群聊的最后一条消息时间
        messages = im_loader.get_session_messages(session_id)
        if not messages.empty:
            last_time = messages['send_time_ms'].max()
            time_str = im_loader.timestamp_to_datetime(last_time)
            print(f"  - {session_id}: 最后活跃 {time_str}")
        else:
            print(f"  - {session_id}: 无消息")

# ============ 群聊活跃度统计 (问题 239-244) ============
print("\n=== 群聊活跃度统计 ===")

def get_group_activity_stats(session_id):
    """获取群聊活跃度统计"""
    messages = im_loader.get_session_messages(session_id)

    if messages.empty:
        return {}

    # 按发送人统计消息数
    sender_counts = messages.groupby('from_user_id').size().sort_values(ascending=False)

    return sender_counts.to_dict()

for session_id, _ in group_sessions:
    stats = get_group_activity_stats(session_id)

    if stats:
        print(f"\n群聊 {session_id} 活跃度排行:")
        for i, (sender_id, count) in enumerate(list(stats.items())[:5], 1):
            sender_name = im_loader.get_user_name(sender_id)
            print(f"  {i}. {sender_name}: {count}条")

# ============ 特定群聊消息 (问题 245-249) ============
print("\n=== 特定群聊消息 ===")

# 由于数据中没有"项目组"群，列出所有群聊的最新消息
print("各群聊最新消息:")

for session_id, _ in group_sessions:
    messages = im_loader.get_session_messages(session_id)

    if not messages.empty:
        latest = messages.sort_values('send_time_ms', ascending=False).iloc[0]
        sender = im_loader.get_user_name(latest['from_user_id'])
        time = im_loader.timestamp_to_datetime(latest['send_time_ms'])
        content = latest['content'][:50] + '...' if len(latest['content']) > 50 else latest['content']

        print(f"\n群聊 {session_id}:")
        print(f"  最新: [{time}] {sender}: {content}")

print("\n" + "=" * 60)
```
