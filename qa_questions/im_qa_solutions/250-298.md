## 问题描述分类
- 问题 250-256: 搜索特定群组（年底大促）
- 问题 257-262: 素材压缩包搜索
- 问题 263-268: 网红相关资料
- 问题 269-275: 会话统计
- 问题 276-282: 消息数量统计
- 问题 283-290: 消息类型和时间分布
- 问题 291-294: 不活跃会话
- 问题 295-298: 消息长度统计

## 代码实现

```python
from utils import im_loader, CURRENT_USER_ID
from datetime import datetime, timedelta
import pandas as pd

print("=" * 60)
print("IM系统查询 - 群组搜索、会话统计、活跃度分析")
print("=" * 60)

# ============ 搜索特定群组 (问题 250-256) ============
print("\n=== 搜索特定群组 ===")

group_keywords = ["年底大促", "大促", "项目组", "运营组"]
my_sessions = im_loader.get_user_sessions(CURRENT_USER_ID)
group_sessions = [(sid, ctype) for sid, ctype in my_sessions if ctype == 3]

print(f"我参与的群聊数量: {len(group_sessions)}")

found_groups = []
for session_id, _ in group_sessions:
    messages = im_loader.get_session_messages(session_id)
    # 检查消息内容是否包含关键词
    for kw in group_keywords:
        matching = messages[messages['content'].str.contains(kw, na=False, case=False)]
        if not matching.empty:
            found_groups.append((session_id, kw))
            break

if found_groups:
    print("\n找到相关群组:")
    for session_id, kw in found_groups:
        print(f"  - {session_id} (关键词: {kw})")
else:
    print("\n没有找到包含相关关键词的群组")

# ============ 素材压缩包搜索 (问题 257-262) ============
print("\n=== 素材压缩包搜索 ===")

received = im_loader.get_user_received_messages(CURRENT_USER_ID)
files = received[received['message_type'] == 3]

# 搜索包含"素材"的文件
material_files = files[files['content'].str.contains('素材', na=False, case=False)]

print(f"素材相关文件: {len(material_files)}个")
for _, msg in material_files.iterrows():
    sender = im_loader.get_user_name(msg['from_user_id'])
    time = im_loader.timestamp_to_datetime(msg['send_time_ms'])
    print(f"  - {msg['content']} (来自:{sender}, 时间:{time})")

# 搜索压缩包
archive_keywords = ['.zip', '.rar', '.7z', '.tar', '.gz']
archives = files[files['content'].str.contains('|'.join(archive_keywords), case=False, na=False)]

print(f"\n压缩包文件: {len(archives)}个")
for _, msg in archives.iterrows():
    sender = im_loader.get_user_name(msg['from_user_id'])
    time = im_loader.timestamp_to_datetime(msg['send_time_ms'])
    print(f"  - {msg['content']} (来自:{sender}, 时间:{time})")

# ============ 网红相关资料 (问题 263-268) ============
print("\n=== 网红相关资料 ===")

# 搜索包含"网红"的消息
my_sessions = im_loader.get_user_sessions(CURRENT_USER_ID)
all_messages = []

for session_id, conv_type in my_sessions:
    messages = im_loader.get_session_messages(session_id)
    matching = messages[messages['content'].str.contains('网红', na=False, case=False)]
    if not matching.empty:
        all_messages.append(matching)

if all_messages:
    result = pd.concat(all_messages)
    print(f"找到 {len(result)} 条关于'网红'的消息")

    # 按发送人统计
    sender_counts = result.groupby('from_user_id').size()
    print("\n按发送人统计:")
    for sender_id, count in sender_counts.items():
        sender_name = im_loader.get_user_name(sender_id)
        print(f"  {sender_name}: {count}条")

    # 列出相关文件
    wanghong_files = result[result['message_type'] == 3]
    if not wanghong_files.empty:
        print("\n网红相关文件:")
        for _, msg in wanghong_files.iterrows():
            sender = im_loader.get_user_name(msg['from_user_id'])
            print(f"  - {msg['content']} (来自:{sender})")
else:
    print("没有找到关于'网红'的消息")

# ============ 会话统计 (问题 269-275) ============
print("\n=== 会话统计 ===")

print(f"\n总会话数: {len(my_sessions)}")

private_chats = [(sid, ctype) for sid, ctype in my_sessions if ctype == 1]
group_chats = [(sid, ctype) for sid, ctype in my_sessions if ctype == 3]

print(f"私聊数: {len(private_chats)}")
print(f"群聊数: {len(group_chats)}")

# 统计每个会话的消息数
session_message_counts = []
for session_id, conv_type in my_sessions:
    messages = im_loader.get_session_messages(session_id)
    count = len(messages)
    session_message_counts.append((session_id, conv_type, count))

# 按消息数排序
session_message_counts.sort(key=lambda x: x[2], reverse=True)

print("\n会话消息数排行:")
for i, (session_id, conv_type, count) in enumerate(session_message_counts[:10], 1):
    ctype = im_loader.CONVERSATION_TYPE.get(conv_type, '未知')
    print(f"  {i}. [{ctype}] {session_id}: {count}条")

# ============ 消息数量统计 (问题 276-282) ============
print("\n=== 消息数量统计 ===")

# 我发送的消息
sent = im_loader.get_user_sent_messages(CURRENT_USER_ID)

# 按时间范围统计
now = datetime.now()
week_ago_ts = int((now - timedelta(days=7)).timestamp() * 1000)
month_ago_ts = int((now - timedelta(days=30)).timestamp() * 1000)

sent_week = sent[sent['send_time_ms'] >= week_ago_ts]
sent_month = sent[sent['send_time_ms'] >= month_ago_ts]

print(f"过去7天发送: {len(sent_week)}条")
print(f"过去30天发送: {len(sent_month)}条")
print(f"总共发送: {len(sent)}条")

# 按日期统计每天的发送量
sent['date'] = sent['send_time_ms'].apply(lambda x: datetime.fromtimestamp(x / 1000).date())
daily_counts = sent.groupby('date').size().sort_index(ascending=False)

print("\n按日期统计发送量 (最近10天):")
for date, count in daily_counts.head(10).items():
    print(f"  {date}: {count}条")

# 与特定用户的消息统计
print("\n与萧翔的消息统计:")
messages_with_xiang = im_loader.get_messages_between_users(CURRENT_USER_ID, 10051)

if len(messages_with_xiang) > 0:
    my_count = len(messages_with_xiang[messages_with_xiang['from_user_id'] == CURRENT_USER_ID])
    his_count = len(messages_with_xiang[messages_with_xiang['from_user_id'] == 10051])

    print(f"  我发送: {my_count}条")
    print(f"  萧翔发送: {his_count}条")
    print(f"  总计: {len(messages_with_xiang)}条")
else:
    print("  无消息记录")

# ============ 消息类型和时间分布 (问题 283-290) ============
print("\n=== 消息类型和时间分布 ===")

# 按消息类型统计
type_counts = sent.groupby('message_type').size()
print("\n按消息类型统计:")
for msg_type, count in type_counts.items():
    type_name = im_loader.MESSAGE_TYPE.get(msg_type, f"类型{msg_type}")
    print(f"  {type_name}: {count}条")

# 按小时统计发送时间分布
sent['hour'] = sent['send_time_ms'].apply(lambda x: datetime.fromtimestamp(x / 1000).hour)
hourly_counts = sent.groupby('hour').size()

print("\n按小时统计发送量:")
for hour in sorted(hourly_counts.index):
    count = hourly_counts[hour]
    print(f"  {hour:02d}:00: {count}条")

most_active_hour = hourly_counts.idxmax()
print(f"\n最活跃时段: {most_active_hour:02d}:00 - {most_active_hour+1:02d}:00")

# ============ 不活跃会话 (问题 291-294) ============
print("\n=== 不活跃会话 ===")

cutoff_ts = int((datetime.now() - timedelta(days=30)).timestamp() * 1000)

inactive_sessions = []
for session_id, conv_type in my_sessions:
    messages = im_loader.get_session_messages(session_id)

    if not messages.empty:
        last_time = messages['send_time_ms'].max()
        if last_time < cutoff_ts:
            days_inactive = (datetime.now().timestamp() * 1000 - last_time) / (24 * 3600 * 1000)
            inactive_sessions.append((session_id, conv_type, days_inactive))

inactive_sessions.sort(key=lambda x: x[2], reverse=True)

print(f"\n超过30天未活跃的会话: {len(inactive_sessions)}个")
for session_id, conv_type, days in inactive_sessions[:10]:
    ctype = im_loader.CONVERSATION_TYPE.get(conv_type, '未知')
    print(f"  - [{ctype}] {session_id}: {days:.0f}天未活跃")

# ============ 消息长度统计 (问题 295-298) ============
print("\n=== 消息长度统计 ===")

sent['content_length'] = sent['content'].str.len()

avg_length = sent['content_length'].mean()
print(f"\n我发送的消息平均长度: {avg_length:.0f}字符")

# 按会话统计平均长度
session_avg_lengths = []
for session_id, conv_type in my_sessions:
    messages = im_loader.get_session_messages(session_id)
    if not messages.empty:
        messages_copy = messages.copy()
        messages_copy['length'] = messages_copy['content'].str.len()
        avg_len = messages_copy['length'].mean()
        session_avg_lengths.append((session_id, conv_type, avg_len))

session_avg_lengths.sort(key=lambda x: x[2], reverse=True)

print("\n会话消息平均长度排行:")
for session_id, conv_type, avg_len in session_avg_lengths[:5]:
    ctype = im_loader.CONVERSATION_TYPE.get(conv_type, '未知')
    print(f"  [{ctype}] {session_id}: {avg_len:.0f}字符")

# 最长的消息
longest_messages = sent.nlargest(5, 'content_length')
print("\n最长的5条消息:")
for i, (_, msg) in enumerate(longest_messages.iterrows(), 1):
    time = im_loader.timestamp_to_datetime(msg['send_time_ms'])
    content = msg['content'][:50] + '...' if len(msg['content']) > 50 else msg['content']
    print(f"  {i}. [{time}] {len(msg['content'])}字符: {content}")

print("\n" + "=" * 60)
```
