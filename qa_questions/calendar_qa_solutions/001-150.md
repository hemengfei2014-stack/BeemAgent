# 问题 1-150: 今天会议、上午/下午/晚上安排、空闲时段

## 问题描述分类
- 问题 1-12: 今天有哪些会议
- 问题 13-18: 上午安排
- 问题 19-30: 下午安排
- 问题 31-36: 晚上安排
- 问题 37-48: 第一场/最后一场会议时间
- 问题 49-60: 会议总数
- 问题 61-72: 空闲时段
- 问题 73-90: 上午/下午/晚上详细安排
- 问题 91-150: 更多时间相关查询

## 代码实现

```python
from utils import calendar_loader, CURRENT_USER_ID
from datetime import date, datetime, timedelta

print("=" * 60)
print(f"日程系统查询 - 当前用户: {calendar_loader.get_user_name(CURRENT_USER_ID)} (ID:{CURRENT_USER_ID})")
print("=" * 60)

# 获取今天的日期
today = date.today()

# ============ 今天的会议 (问题 1-12) ============
print(f"\n【问题 1-12】今天的会议安排 ({today})")

today_events = calendar_loader.get_events_by_date(CURRENT_USER_ID, today)

if today_events:
    print(f"\n今天共有 {len(today_events)} 场会议:\n")

    for i, event in enumerate(today_events, 1):
        print(f"{i}. {event.subject}")
        print(f"   时间: {calendar_loader.timestamp_to_time(event.start_time)} - {calendar_loader.timestamp_to_time(event.end_time)}")
        print(f"   组织者: {calendar_loader.get_user_name(event.organizer_id)}")
        participants = calendar_loader.get_event_participants(event.event_id)
        print(f"   参与者: {len(participants)}人")
else:
    print(f"\n今天没有会议安排")

# ============ 上午/下午/晚上安排 (问题 13-36) ============
print("\n=== 按时段划分的安排 ===")

periods = {
    '上午': 'morning',
    '下午': 'afternoon',
    '晚上': 'evening'
}

for period_name, period_key in periods.items():
    period_events = calendar_loader.get_events_by_period(CURRENT_USER_ID, today, period_name)

    print(f"\n【{period_name}】")
    if period_events:
        print(f"  共 {len(period_events)} 场会议:")
        for event in period_events:
            time_str = calendar_loader.timestamp_to_time(event.start_time)
            print(f"  - {time_str} {event.subject}")
    else:
        print("  无会议安排")

# ============ 第一场/最后一场会议 (问题 37-48) ============
print("\n=== 第一场/最后一场会议 ===")

if today_events:
    # 按开始时间排序
    sorted_events = sorted(today_events, key=lambda x: x.start_time)

    first = sorted_events[0]
    last = sorted_events[-1]

    print(f"\n第一场会议:")
    print(f"  时间: {calendar_loader.timestamp_to_time(first.start_time)}")
    print(f"  主题: {first.subject}")

    print(f"\n最后一场会议:")
    print(f"  时间: {calendar_loader.timestamp_to_time(last.end_time)}")
    print(f"  主题: {last.subject}")
else:
    print("\n今天没有会议")

# ============ 会议总数 (问题 49-60) ============
print("\n=== 会议统计 ===")

stats = calendar_loader.get_event_stats(CURRENT_USER_ID, today)

print(f"\n今天会议总数: {stats['total']}场")
print(f"  上午: {stats['morning']}场")
print(f"  下午: {stats['afternoon']}场")
print(f"  晚上: {stats['evening']}场")

# ============ 空闲时段 (问题 61-72) ============
print("\n=== 空闲时段 ===")

free_slots = calendar_loader.get_free_time_slots(CURRENT_USER_ID, today)

if free_slots:
    print("\n今天的空闲时段:")
    for i, slot in enumerate(free_slots, 1):
        start_str = datetime.fromtimestamp(slot['start'] / 1000).strftime('%H:%M')
        end_str = datetime.fromtimestamp(slot['end'] / 1000).strftime('%H:%M')
        duration_min = slot['duration'] // (60 * 1000)
        print(f"  {i}. {start_str} - {end_str} ({duration_min}分钟)")
else:
    print("\n今天没有空闲时段")

# ============ 详细日程表 ============
print("\n=== 今天完整日程表 ===")

if today_events:
    print(f"\n日期: {today}")
    print(f"会议数量: {len(today_events)}场\n")

    # 按时间排序
    sorted_events = sorted(today_events, key=lambda x: x.start_time)

    for i, event in enumerate(sorted_events, 1):
        start_str = calendar_loader.timestamp_to_datetime(event.start_time)
        end_str = calendar_loader.timestamp_to_time(event.end_time)
        print(f"{i}. [{start_str} - {end_str}] {event.subject}")
        print(f"   组织者: {calendar_loader.get_user_name(event.organizer_id)}")
else:
    print(f"\n{today} 无会议安排")

print("\n" + "=" * 60)
```
