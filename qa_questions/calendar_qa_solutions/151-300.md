# 问题 151-300: 未来会议、会议详情、搜索

## 问题描述分类
- 问题 151-180: 未来N天的会议
- 问题 181-210: 特定会议详情
- 问题 211-240: 按关键词搜索会议
- 问题 241-270: 与某人的会议
- 问题 271-300: 会议统计

## 代码实现

```python
from utils import calendar_loader, CURRENT_USER_ID
from datetime import date, datetime, timedelta

print("=" * 60)
print("日程系统查询 - 未来会议、会议详情、搜索")
print("=" * 60)

# ============ 未来N天的会议 (问题 151-180) ============
print("\n=== 未来N天的会议 ===")

today = date.today()

# 查看未来7天的会议
print("\n未来7天的会议:")
for i in range(7):
    check_date = today + timedelta(days=i)
    events = calendar_loader.get_events_by_date(CURRENT_USER_ID, check_date)

    if events:
        print(f"\n{check_date.strftime('%Y-%m-%d (%A)')} - {len(events)}场会议:")
        for event in events:
            time_str = calendar_loader.timestamp_to_time(event.start_time)
            print(f"  - {time_str} {event.subject}")
    else:
        print(f"\n{check_date.strftime('%Y-%m-%d')} - 无会议")

# ============ 我参与的所有会议 (问题 181-210) ============
print("\n\n=== 我参与的所有会议 ===")

all_events = calendar_loader.get_user_events(CURRENT_USER_ID)

if all_events:
    print(f"\n共参与 {len(all_events)} 场会议")

    # 按时间排序（从未来到过去）
    now_ts = int(datetime.now().timestamp() * 1000)
    future_events = [e for e in all_events if e.start_time >= now_ts]
    past_events = [e for e in all_events if e.start_time < now_ts]

    print(f"\n未来会议: {len(future_events)}场")
    if future_events:
        for event in sorted(future_events, key=lambda x: x.start_time)[:10]:
            date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
            time_str = calendar_loader.timestamp_to_time(event.start_time)
            print(f"  - [{date_str} {time_str}] {event.subject}")

    print(f"\n历史会议: {len(past_events)}场")
    if past_events:
        for event in sorted(past_events, key=lambda x: x.start_time, reverse=True)[:5]:
            date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
            time_str = calendar_loader.timestamp_to_time(event.start_time)
            print(f"  - [{date_str} {time_str}] {event.subject}")
else:
    print("\n没有参与任何会议")

# ============ 关键词搜索会议 (问题 211-240) ============
print("\n\n=== 关键词搜索会议 ===")

keywords = ["流量", "带货", "直播", "短视频", "推荐", "滤镜", "粉丝", "剪辑", "特效", "拍摄"]

for keyword in keywords:
    matching = []
    for event in all_events:
        if keyword in event.subject:
            matching.append(event)

    if matching:
        print(f"\n关键词'{keyword}': {len(matching)}场会议")
        for event in matching[:3]:
            date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
            time_str = calendar_loader.timestamp_to_time(event.start_time)
            print(f"  - [{date_str} {time_str}] {event.subject}")
            print(f"    组织者: {calendar_loader.get_user_name(event.organizer_id)}")

# ============ 与特定人的会议 (问题 241-270) ============
print("\n\n=== 与特定人的共同会议 ===")

# 查找和萧翔（10051）的共同会议
target_user_id = 10051
target_name = "萧翔"

common_events = []
for event in all_events:
    participants = calendar_loader.get_event_participants(event.event_id)
    if CURRENT_USER_ID in participants and target_user_id in participants:
        common_events.append(event)

if common_events:
    print(f"\n和{target_name}的共同会议: {len(common_events)}场")
    for event in common_events:
        date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
        time_str = calendar_loader.timestamp_to_time(event.start_time)
        print(f"  - [{date_str} {time_str}] {event.subject}")
else:
    print(f"\n和{target_name}没有共同会议")

# ============ 会议统计 (问题 271-300) ============
print("\n\n=== 会议统计 ===")

if all_events:
    # 按日期统计
    date_counts = {}
    for event in all_events:
        event_date = datetime.fromtimestamp(event.start_time / 1000).date()
        date_counts[event_date] = date_counts.get(event_date, 0) + 1

    print(f"\n按日期统计 (会议最多的前5天):")
    sorted_dates = sorted(date_counts.items(), key=lambda x: x[1], reverse=True)[:5]
    for event_date, count in sorted_dates:
        print(f"  {event_date}: {count}场会议")

    # 作为组织者 vs 参与者
    organized = [e for e in all_events if e.organizer_id == CURRENT_USER_ID]
    participated = [e for e in all_events if e.organizer_id != CURRENT_USER_ID]

    print(f"\n作为组织者: {len(organized)}场")
    print(f"作为参与者: {len(participated)}场")

    # 会议时长统计
    durations = [(e.end_time - e.start_time) // (60 * 1000) for e in all_events]
    if durations:
        avg_duration = sum(durations) // len(durations)
        print(f"\n平均会议时长: {avg_duration}分钟")
        print(f"最短: {min(durations)}分钟")
        print(f"最长: {max(durations)}分钟")

print("\n" + "=" * 60)
```
