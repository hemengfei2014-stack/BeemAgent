# 问题 1-100: 组织+消息（同事信息+聊天记录）

## 问题描述分类
- 问题 1-16: 萧翔的信息+联系方式+聊天记录
- 问题 17-30: 李世的信息+邮箱+聊天记录
- 问题 31-44: 贝冠的信息+联系方式+聊天记录
- 问题 45-69: 我部门同事+群聊讨论
- 问题 70-77: 我的直属领导+消息
- 问题 78-84: 我的下属+消息

## 代码实现

```python
from utils import cross_loader, CURRENT_USER_ID
from datetime import datetime

print("=" * 60)
print("跨系统查询 - 组织信息 + IM聊天记录")
print("=" * 60)

# 当前用户
current_user = cross_loader.get_user_profile(CURRENT_USER_ID)
print(f"\n当前用户: {current_user['name']} (ID:{current_user['id']})")

# ============ 萧翔的信息 + 聊天记录 (问题 1-16) ============
print("\n=== 萧翔 ===")

# 查找萧翔
target_name = "萧翔"
users = cross_loader.org_loader.get_user_by_name(target_name, CURRENT_ORG_ID)

if users:
    xiang = users[0]
    print(f"\n姓名: {xiang.name}")
    print(f"工号: {xiang.id}")

    # 部门信息
    dept_info = cross_loader.get_user_dept_info(xiang.id)
    print(f"部门: {dept_info.get('dept_name', '无')}")

    # 联系方式
    contact = cross_loader.get_user_contact_info(xiang.id)
    print(f"手机: {contact.get('mobile', '无')}")
    print(f"邮箱: {contact.get('email', '无')}")

    # 聊天记录
    print(f"\n与{current_user['name']}的聊天记录:")
    messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, xiang.id)

    if len(messages) > 0:
        print(f"共 {len(messages)} 条消息")

        # 最近的消息
        recent = messages.sort_values('send_time_ms', ascending=False).head(5)
        for _, msg in recent.iterrows():
            time = cross_loader.im_loader.timestamp_to_datetime(msg['send_time_ms'])
            sender = "我" if msg['from_user_id'] == CURRENT_USER_ID else xiang.name
            content = msg['content'][:40] + "..." if len(msg['content']) > 40 else msg['content']
            print(f"  [{time}] {sender}: {content}")
    else:
        print("没有聊天记录")
else:
    print(f"\n未找到用户: {target_name}")

# ============ 李世的信息 + 邮箱 + 聊天记录 (问题 17-30) ============
print("\n\n=== 李世 ===")

target_name = "李世"
users = cross_loader.org_loader.get_user_by_name(target_name, CURRENT_ORG_ID)

if users:
    li = users[0]
    print(f"\n姓名: {li.name}")
    print(f"工号: {li.id}")

    # 邮箱
    print(f"邮箱: {li.email}")

    # 聊天记录
    print(f"\n与{current_user['name']}的聊天记录:")
    messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, li.id)

    if len(messages) > 0:
        print(f"共 {len(messages)} 条消息")
    else:
        print("没有聊天记录")
else:
    print(f"\n未找到用户: {target_name}")

# ============ 贝冠的信息 + 联系方式 + 聊天记录 (问题 31-44) ============
print("\n\n=== 贝冠 ===")

target_name = "贝冠"
users = cross_loader.org_loader.get_user_by_name(target_name, CURRENT_ORG_ID)

if users:
    bei = users[0]
    print(f"\n姓名: {bei.name}")
    print(f"工号: {bei.id}")

    # 联系方式
    print(f"手机: {bei.mobile}")
    print(f"邮箱: {bei.email}")

    # 聊天记录
    print(f"\n与{current_user['name']}的聊天记录:")
    messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, bei.id)

    if len(messages) > 0:
        print(f"共 {len(messages)} 条消息")
    else:
        print("没有聊天记录")
else:
    print(f"\n未找到用户: {target_name}")

# ============ 我的部门同事 + 群聊讨论 (问题 45-69) ============
print("\n\n=== 我的部门同事 ===")

my_dept = cross_loader.org_loader.get_user_dept(CURRENT_USER_ID)
if my_dept is not None:
    print(f"\n部门: {my_dept['name']}")

    # 部门成员
    dept_members = cross_loader.get_dept_members(my_dept['id'])
    print(f"\n部门成员 ({len(dept_members)}人):")
    for member in dept_members:
        print(f"  - {member.name} (工号:{member.id})")

    # 群聊讨论（查询群聊中的关键词）
    print("\n群聊讨论话题:")
    group_keywords = ["流量", "带货", "直播", "短视频"]

    my_sessions = cross_loader.im_loader.get_user_sessions(CURRENT_USER_ID)
    group_sessions = [s for s in my_sessions if s[1] == 3]  # 3=群聊

    if group_sessions:
        print(f"  我参与的群聊: {len(group_sessions)}个")

        for kw in group_keywords:
            for session_id, _ in group_sessions:
                messages = cross_loader.im_loader.get_session_messages(session_id)
                matching = messages[messages['content'].str.contains(kw, na=False, case=False)]
                if len(matching) > 0:
                    print(f"  - '{kw}': {len(matching)}条消息")
                    break
    else:
        print("  未参与任何群聊")
else:
    print("\n未分配部门")

# ============ 我的直属领导 + 消息 (问题 70-77) ============
print("\n\n=== 我的直属领导 ===")

manager_info = cross_loader.get_user_manager_info(CURRENT_USER_ID)

if manager_info:
    print(f"\n领导: {manager_info['manager_name']}")
    print(f"工号: {manager_info['manager_id']}")
    print(f"职位: {manager_info['manager_role']}")

    # 领导发来的消息
    messages = cross_loader.get_recent_messages_from_user(manager_info['manager_id'], CURRENT_USER_ID, 5)

    if len(messages) > 0:
        print(f"\n最近发来的消息:")
        for _, msg in messages.iterrows():
            time = cross_loader.im_loader.timestamp_to_datetime(msg['send_time_ms'])
            content = msg['content'][:40] + "..." if len(msg['content']) > 40 else msg['content']
            print(f"  [{time}] {content}")
    else:
        print("\n没有发过消息")
else:
    print("\n没有直属领导")

# ============ 我的下属 + 消息 (问题 78-84) ============
print("\n\n=== 我的下属 ===")

subordinates = cross_loader.get_team_members(CURRENT_USER_ID)

if subordinates:
    print(f"\n直属下属: {len(subordinates)}人")
    for sub in subordinates:
        print(f"  - {sub.name} (工号:{sub.id})")

        # 检查是否有聊天记录
        messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, sub.id)
        if len(messages) > 0:
            latest_time = cross_loader.im_loader.timestamp_to_datetime(
                messages['send_time_ms'].max()
            )
            print(f"    最近聊天: {latest_time}")
        else:
            print(f"    无聊天记录")
else:
    print("\n没有下属")

print("\n" + "=" * 60)
```
