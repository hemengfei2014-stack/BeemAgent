# 问题 201-300: 跨系统查询（用户信息+文档+会议）

## 问题描述分类
- 问题 201-214: 萧翔的详细信息+文档
- 问题 215-220: 张姓同事的文档
- 问题 221-230: 研发部负责人+文档
- 问题 231-245: 部门成员的文档
- 问题 246-260: 特定用户特定类型文档
- 问题 261-270: 文档创建时间统计
- 问题 271-300: 共同会议+聊天记录

## 代码实现

```python
from utils import cross_loader, CURRENT_USER_ID
from datetime import datetime

print("=" * 60)
print("跨系统查询 - 用户详情+文档+会议")
print("=" * 60)

# ============ 萧翔的详细信息 (问题 201-214) ============
print("\n=== 萧翔详细信息 ===")

xiang_id = 10051
xiang = cross_loader.get_user_profile(xiang_id)

if xiang is not None:
    print(f"\n姓名: {xiang['name']}")
    print(f"工号: {xiang['id']}")
    print(f"性别: {cross_loader.org_loader.get_gender_name(xiang['gender'])}")
    print(f"入职时间: {cross_loader.doc_loader.timestamp_to_date(xiang['join_date'])}")

    # 部门信息
    dept_info = cross_loader.get_user_dept_info(xiang_id)
    print(f"部门: {dept_info.get('dept_name', '无')}")

    # 创建的文档
    xiang_docs = cross_loader.doc_loader.get_user_created_docs(xiang_id)
    print(f"\n创建的文档: {len(xiang_docs)}个")
    for doc in xiang_docs:
        print(f"  - {doc.title} ({cross_loader.doc_loader.get_doc_type_name(doc.ctype)})")

    # 我能访问的
    my_accessible = set(cross_loader.doc_loader.get_user_accessible_docs(CURRENT_USER_ID))
    accessible = [d for d in xiang_docs if d.doc_id in my_accessible]
    print(f"\n我能访问的: {len(accessible)}个")
    for doc in accessible:
        print(f"  - {doc.title}")

# ============ 张姓同事的文档 (问题 215-220) ============
print("\n\n=== 张姓同事的文档 ===")

zhang_users = cross_loader.org_loader.get_users_by_name_pattern("张", CURRENT_ORG_ID)

print(f"\n张姓同事: {len(zhang_users)}人")
for user in zhang_users:
    docs = cross_loader.doc_loader.get_user_created_docs(user.id)
    my_accessible = set(cross_loader.doc_loader.get_user_accessible_docs(CURRENT_USER_ID))

    # 我能访问的
    accessible = [d for d in docs if d.doc_id in my_accessible]

    if accessible:
        print(f"\n{user.name} - 我能访问的文档:")
        for doc in accessible:
            print(f"  - {doc.title}")

# ============ 研发部负责人+文档 (问题 221-230) ============
print("\n\n=== 研发部（类似部门）负责人+文档 ===")

# 查找"研发"或类似部门
depts = cross_loader.org_loader.get_dept_by_name("研发", CURRENT_ORG_ID)
# 也检查"后端"等部门
all_depts = cross_loader.org_loader.get_org_depts(CURRENT_ORG_ID)

rd_like_depts = [d for d in all_depts if "研发" in d.name or "后端" in d.name]

print(f"\n研发相关部门:")
for dept in rd_like_depts:
    print(f"  - {dept.name} (负责人:{cross_loader.org_loader.get_user_name(int(dept.leader_id)) if dept.leader_id else '无'})")

    # 负责人创建的文档
    if dept.leader_id:
        leader_id = int(dept.leader_id)
        leader_docs = cross_loader.doc_loader.get_user_created_docs(leader_id)
        print(f"    负责人创建的文档: {len(leader_docs)}个")
        for doc in leader_docs:
            print(f"      - {doc.title}")

# ============ 部门成员的文档 (问题 231-245) ============
print("\n\n=== 我部门成员的文档 ===")

my_dept = cross_loader.org_loader.get_user_dept(CURRENT_USER_ID)
if my_dept is not None:
    print(f"\n我的部门: {my_dept['name']}")

    dept_members = cross_loader.get_dept_members(my_dept['id'])
    my_accessible = set(cross_loader.doc_loader.get_user_accessible_docs(CURRENT_USER_ID))

    print(f"\n部门成员创建的、我能访问的文档:")
    for member in dept_members:
        if member.id == CURRENT_USER_ID:
            continue

        docs = cross_loader.doc_loader.get_user_created_docs(member.id)
        accessible = [d for d in docs if d.doc_id in my_accessible]

        if accessible:
            print(f"\n{member.name}:")
            for doc in accessible:
                print(f"  - {doc.title}")

# ============ 特定用户特定类型文档 (问题 246-260) ============
print("\n\n=== 特定用户特定类型文档 ===")

# 李世创建的关于"流量"的文档
li_id = 10008
li_docs = cross_loader.doc_loader.get_user_created_docs(li_id)

print(f"\n李世创建的、标题含'流量'的文档:")
for doc in li_docs:
    if "流量" in doc.title:
        print(f"  - {doc.title}")

# 萧翔创建的PDF文档
xiang_id = 10051
xiang_docs = cross_loader.doc_loader.get_user_created_docs(xiang_id)

pdf_docs = [d for d in xiang_docs if d.ctype == 'pdf']
print(f"\n萧翔创建的PDF文档: {len(pdf_docs)}个")
for doc in pdf_docs:
    print(f"  - {doc.title}")

# ============ 文档创建时间统计 (问题 261-270) ============
print("\n\n=== 文档创建时间统计 ===")

accessible_docs = cross_loader.doc_loader.get_user_docs(CURRENT_USER_ID)

if accessible_docs:
    # 按创建者统计
    creator_times = {}
    for doc in accessible_docs:
        if doc.owner not in creator_times:
            creator_times[doc.owner] = []
        creator_times[doc.owner].append(doc.create_time)

    print("\n各创建者的文档创建时间:")
    for creator_id, times in creator_times.items():
        creator_name = cross_loader.org_loader.get_user_name(creator_id)
        earliest = min(times)
        latest = max(times)
        print(f"  {creator_name}: {cross_loader.doc_loader.timestamp_to_date(earliest)} - {cross_loader.doc_loader.timestamp_to_date(latest)}")

# ============ 共同会议+聊天记录 (问题 271-300) ============
print("\n\n=== 共同会议+聊天记录 ===")

users_to_check = [
    (10008, "李世"),
    (10051, "萧翔"),
    (10044, "贝冠"),
]

for target_id, target_name in users_to_check:
    print(f"\n=== 和{target_name} ===")

    # 共同会议
    my_events = cross_loader.calendar_loader.get_user_events(CURRENT_USER_ID)
    target_events = cross_loader.calendar_loader.get_user_events(target_id)

    my_event_ids = set([e.event_id for e in my_events])
    target_event_ids = set([e.event_id for e in target_events])

    common_event_ids = my_event_ids & target_event_ids
    common_events = [e for e in my_events if e.event_id in common_event_ids]

    if common_events:
        print(f"\n共同会议: {len(common_events)}场")
        for event in common_events:
            date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
            time_str = cross_loader.calendar_loader.timestamp_to_time(event.start_time)
            print(f"  - [{date_str} {time_str}] {event.subject}")
    else:
        print(f"\n共同会议: 0场")

    # 聊天记录
    messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, target_id)
    print(f"聊天记录: {len(messages)}条")

print("\n" + "=" * 60)
```
