# 问题 101-200: 跨系统查询（组织+消息+文档）

## 问题描述分类
- 问题 101-106: 产品经理 + 聊天频率
- 问题 107-121: 特定用户最新消息
- 问题 122-141: 发消息最多的同事统计
- 问题 142-156: 部门成员之间的聊天
- 问题 157-187: 共享文档查询
- 问题 188-200: 萧翔/MCN相关文档

## 代码实现

```python
from utils import cross_loader, CURRENT_USER_ID

print("=" * 60)
print("跨系统查询 - 消息统计、共享文档")
print("=" * 60)

# ============ 产品经理 + 聊天频率 (问题 102-106) ============
print("\n=== 产品经理同事 + 聊天频率 ===")

# 获取产品经理
ROLE_PRODUCT_MANAGER = 1
pms = cross_loader.org_loader.get_users_by_role(ROLE_PRODUCT_MANAGER, CURRENT_ORG_ID)

print(f"\n产品经理: {len(pms)}人")

chat_counts = {}
for pm in pms:
    messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, pm.id)
    if len(messages) > 0:
        chat_counts[pm.name] = len(messages)
        print(f"  - {pm.name}: {len(messages)}条消息")

if chat_counts:
    most_chatted = max(chat_counts, key=chat_counts.get)
    print(f"\n和我聊天最频繁的产品经理: {most_chatted} ({chat_counts[most_chatted]}条)")
else:
    print("\n没有和产品经理聊过天")

# ============ 特定用户最新消息 (问题 107-121) ============
print("\n\n=== 特定用户最新消息 ===")

users_to_check = [
    (10051, "萧翔"),
    (10008, "李世"),
    (10044, "贝冠"),
]

for user_id, name in users_to_check:
    print(f"\n{name}:")

    messages = cross_loader.get_recent_messages_from_user(user_id, CURRENT_USER_ID, 3)

    if len(messages) > 0:
        print(f"  最近消息: {len(messages)}条")
        for _, msg in messages.iterrows():
            time = cross_loader.im_loader.timestamp_to_datetime(msg['send_time_ms'])
            content = msg['content'][:50] + "..." if len(msg['content']) > 50 else msg['content']
            print(f"  [{time}] {content}")
    else:
        print("  没有发过消息")

# ============ 发消息最多的同事 (问题 122-141) ============
print("\n\n=== 谁给我发消息最多 ===")

# 获取所有联系人及其消息数量
all_senders = cross_loader.im_loader.get_message_count_by_sender(CURRENT_USER_ID)

if all_senders:
    print("\n发消息最多的同事 (Top 5):")
    sorted_senders = sorted(all_senders.items(), key=lambda x: x[1], reverse=True)[:5]

    for sender_id, count in sorted_senders:
        sender_name = cross_loader.org_loader.get_user_name(sender_id)
        print(f"  {sender_name}: {count}条")
else:
    print("\n没有人发过消息")

# ============ 上级/下属的消息统计 (问题 126-141) ============
print("\n\n=== 上级/下属消息统计 ===")

# 上级
current_user = cross_loader.get_user_profile(CURRENT_USER_ID)
if current_user['manager_id']:
    manager_id = int(current_user['manager_id'])
    manager_name = cross_loader.org_loader.get_user_name(manager_id)

    messages = cross_loader.get_chat_history_between_users(manager_id, CURRENT_USER_ID)
    print(f"\n上级 {manager_name}: {len(messages)}条消息")
else:
    print("\n没有上级")

# 下属
subordinates = cross_loader.get_team_members(CURRENT_USER_ID)
print(f"\n下属消息统计:")

for sub in subordinates:
    messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, sub.id)
    if len(messages) > 0:
        print(f"  {sub.name}: {len(messages)}条")

# ============ 共享文档查询 (问题 157-187) ============
print("\n\n=== 共享文档 ===")

# 和贝冠的共同文档
target_id = 10044
target_name = "贝冠"

shared_docs = cross_loader.get_shared_docs(CURRENT_USER_ID, target_id)

print(f"\n和{target_name}共享的文档: {len(shared_docs)}个")
for doc in shared_docs:
    print(f"  - {doc.title} ({cross_loader.doc_loader.get_doc_type_name(doc.ctype)})")

# 我创建的文档，谁能访问
my_created = cross_loader.doc_loader.get_user_created_docs(CURRENT_USER_ID)

if my_created:
    print(f"\n我创建的文档: {len(my_created)}个")

    for doc in my_created:
        # 查找能访问此文档的用户
        accessible_users = []
        for user in cross_loader.org_loader.get_org_users(CURRENT_ORG_ID):
            if user.id != CURRENT_USER_ID:
                accessible = cross_loader.doc_loader.get_user_accessible_docs(user.id)
                if doc.doc_id in accessible:
                    accessible_users.append(user.name)

        if accessible_users:
            print(f"  - {doc.title}")
            print(f"    可访问: {', '.join(accessible_users[:5])}")
            if len(accessible_users) > 5:
                print(f"    等{len(accessible_users)}人")
else:
    print("\n我没有创建任何文档")

# 萧翔创建的文档
xiang_id = 10051
xiang_docs = cross_loader.doc_loader.get_user_created_docs(xiang_id)

print(f"\n萧翔创建的文档: {len(xiang_docs)}个")
for doc in xiang_docs:
    print(f"  - {doc.title}")

# 我能访问吗
my_accessible = set(cross_loader.doc_loader.get_user_accessible_docs(CURRENT_USER_ID))
accessible = [d for d in xiang_docs if d.doc_id in my_accessible]

print(f"\n我能访问的萧翔文档: {len(accessible)}个")
for doc in accessible:
    print(f"  - {doc.title}")

# ============ 关键词相关文档 (问题 186-200) ============
print("\n\n=== 关键词相关文档 ===")

# 搜索涨粉相关文档
keyword = "涨粉"
docs = cross_loader.doc_loader.search_docs_by_keyword(keyword, CURRENT_USER_ID)

print(f"\n'{keyword}'相关文档: {len(docs)}个")
for doc in docs:
    creator = cross_loader.org_loader.get_user_name(doc.owner)
    print(f"  - {doc.title} (创建者:{creator})")

# 搜索MCN相关文档
keyword = "MCN"
docs = cross_loader.doc_loader.search_docs_by_keyword(keyword, CURRENT_USER_ID)

print(f"\n'{keyword}'相关文档: {len(docs)}个")
for doc in docs:
    creator = cross_loader.org_loader.get_user_name(doc.owner)
    print(f"  - {doc.title} (创建者:{creator})")

print("\n" + "=" * 60)
```
