# 问题 301-383: 多系统组合查询（会议+聊天+文档综合）

## 问题描述分类
- 问题 301-313: 特定用户发起的会议+部门信息
- 问题 314-330: 聊天记录中提到的文档
- 问题 331-338: 会议参会人员的聊天情况
- 问题 339-350: 跨系统综合查询
- 问题 351-383: 四系统组合查询

## 代码实现

```python
from utils import cross_loader, CURRENT_USER_ID
from datetime import datetime

print("=" * 60)
print("跨系统查询 - 多系统组合")
print("=" * 60)

# ============ 特定用户发起的会议 (问题 288-313) ============
print("\n=== 特定用户发起的会议 ===")

users_to_check = [
    (10008, "李世"),
    (10051, "萧翔"),
]

for target_id, target_name in users_to_check:
    print(f"\n{target_name}:")

    # 部门信息
    dept_info = cross_loader.get_user_dept_info(target_id)
    print(f"  部门: {dept_info.get('dept_name', '无')}")

    # 发起的会议
    all_events = cross_loader.calendar_loader.get_user_events(target_id)
    organized = [e for e in all_events if e.organizer_id == target_id]

    print(f"  发起的会议: {len(organized)}场")
    for event in organized[:5]:
        date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
        time_str = cross_loader.calendar_loader.timestamp_to_time(event.start_time)
        print(f"    - [{date_str} {time_str}] {event.subject}")

        # 我是否参与
        participants = cross_loader.calendar_loader.get_event_participants(event.event_id)
        if CURRENT_USER_ID in participants:
            print(f"      (我参与了)")
        else:
            print(f"      (我未参与)")

# ============ 我的上级发起的会议 (问题 308-313) ============
print("\n\n=== 我的上级发起的会议 ===")

current_user = cross_loader.get_user_profile(CURRENT_USER_ID)
if current_user['manager_id']:
    manager_id = int(current_user['manager_id'])
    manager_name = cross_loader.org_loader.get_user_name(manager_id)

    print(f"\n我的上级: {manager_name}")

    # 上级发起的会议
    all_events = cross_loader.calendar_loader.get_user_events(manager_id)
    organized = [e for e in all_events if e.organizer_id == manager_id]

    print(f"发起的会议: {len(organized)}场")
    for event in organized:
        date_str = datetime.fromtimestamp(event.start_time / 1000).strftime('%Y-%m-%d')
        time_str = cross_loader.calendar_loader.timestamp_to_time(event.start_time)
        print(f"  - [{date_str} {time_str}] {event.subject}")

# ============ 聊天记录中提到的文档 (问题 316-330) ============
print("\n\n=== 聊天中的文档关键词 ===")

# 常见的文档关键词
doc_keywords = ["文档", "PDF", "文件", "附件", "需求文档", "报告"]

users_to_check = [
    (10008, "李世"),
    (10001, "马云"),  # 我的上级
    (10051, "萧翔"),
]

for target_id, target_name in users_to_check:
    print(f"\n{target_name}发来的消息中提到文档:")

    messages = cross_loader.get_recent_messages_from_user(target_id, CURRENT_USER_ID, 50)

    found = False
    for _, msg in messages.iterrows():
        for kw in doc_keywords:
            if kw in msg['content']:
                time = cross_loader.im_loader.timestamp_to_datetime(msg['send_time_ms'])
                content = msg['content'][:60] + "..." if len(msg['content']) > 60 else msg['content']
                print(f"  [{time}] {content}")
                found = True
                break

    if not found:
        print("  未找到文档相关内容")

# ============ 会议参会人员的聊天情况 (问题 331-338) ============
print("\n\n=== 会议参会人员的聊天情况 ===")

my_events = cross_loader.calendar_loader.get_user_events(CURRENT_USER_ID)

print(f"\n我参与的会议: {len(my_events)}场")

# 找出和参会人员有聊天的会议
chatted_colleagues = set()

for event in my_events:
    participants = cross_loader.calendar_loader.get_event_participants(event.event_id)

    for participant_id in participants:
        if participant_id == CURRENT_USER_ID:
            continue

        # 检查是否有聊天记录
        messages = cross_loader.get_chat_history_between_users(CURRENT_USER_ID, participant_id)
        if len(messages) > 0:
            participant_name = cross_loader.org_loader.get_user_name(participant_id)
            chatted_colleagues.add(participant_name)

if chatted_colleagues:
    print(f"\n和我有聊过的参会同事: {len(chatted_colleagues)}人")
    for name in chatted_colleagues:
        print(f"  - {name}")
else:
    print("\n没有和参会同事聊过天")

# ============ 多系统综合查询 (问题 339-383) ============
print("\n\n=== 多系统综合查询 ===")

# 既开会又聊天的同事
print("\n既和我一起开过会，又在群里聊过天的同事:")

# 获取所有有聊天记录的用户
chatted_users = set()
for session_id, _ in cross_loader.im_loader.get_user_sessions(CURRENT_USER_ID):
    if session_id.count('_') == 1:  # 私聊
        parts = session_id.split('_')
        chatted_users.add(int(parts[0]))
        chatted_users.add(int(parts[1]))

# 和我一起开过会的用户
met_in_meeting = set()
for event in my_events:
    participants = cross_loader.calendar_loader.get_event_participants(event.event_id)
    met_in_meeting.update(participants)

# 交集（排除自己）
both = chatted_users & met_in_meeting - {CURRENT_USER_ID}

if both:
    print(f"{len(both)}人")
    for user_id in list(both)[:10]:
        user_name = cross_loader.org_loader.get_user_name(user_id)
        print(f"  - {user_name}")
else:
    print("无")

# 参会人员中创建了我能访问文档的
print("\n参会人员中创建了我能访问文档的同事:")

my_accessible = set(cross_loader.doc_loader.get_user_accessible_docs(CURRENT_USER_ID))

doc_creators = set()
for doc_id in my_accessible:
    doc = cross_loader.doc_loader.get_doc_by_id(doc_id)
    if doc is not None:
        doc_creators.add(doc.owner)

# 交集
creators_in_meetings = doc_creators & met_in_meeting

if creators_in_meetings:
    print(f"{len(creators_in_meetings)}人")
    for creator_id in list(creators_in_meetings)[:10]:
        creator_name = cross_loader.org_loader.get_user_name(creator_id)
        print(f"  - {creator_name}")
else:
    print("无")

# ============ 四系统组合 (问题 379-383) ============
print("\n\n=== 四系统组合查询 ===")

# 会议参会人员中：创建了我能访问的文档 + 和我有聊天
combined = creators_in_meetings & chatted_users

if combined:
    print(f"符合条件（开会+创建文档+聊天）: {len(combined)}人")
    for user_id in list(combined)[:10]:
        user_name = cross_loader.org_loader.get_user_name(user_id)
        print(f"  - {user_name}")
else:
    print("符合条件（开会+创建文档+聊天）: 无")

print("\n" + "=" * 60)
```
